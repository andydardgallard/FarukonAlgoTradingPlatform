15/// Converts a string representation of a date and time into a `chrono::DateTime<chrono::Utc>`.
16/// This function uses `chrono::NaiveDateTime::parse_from_str` to parse the input string according to the provided format,
17/// then converts the resulting `NaiveDateTime` to a `DateTime<Utc>`.
18///
19/// # Arguments
20/// * `string` - The date-time string to parse (e.g., "2025-07-08 15:30:00").
21/// * `format` - The expected format of the input string (e.g., "%Y-%m-%d %H:%M:%S").
22///
23/// # Returns
24/// * `anyhow::Result<chrono::DateTime<chrono::Utc>>` - The parsed UTC date-time on success, or an error if parsing fails.
25pub fn string_to_date_time(string: &String, format: &str) -> anyhow::Result<chrono::DateTime<chrono::Utc>> {
26    // Format "%Y-%m-%d %H:%M:%S"
27    let dt = chrono::NaiveDateTime::parse_from_str(
28        string,
29        format,
30    ).with_context(|| format!(
31        "Invalid format '{}'",
32        format
33    ))?;
34
35    let dt_utc = chrono::DateTime::<chrono::Utc>::from_naive_utc_and_offset(
36        dt,
37        chrono::Utc
38    );
39
40    anyhow::Ok(dt_utc)
41}

43/// Calculates the maximum available quantity to trade based on available capital.
44/// # Arguments
45/// * `cash` - Available cash.
46/// * `quantity` - Desired quantity.
47/// * `instrument_info` - Instrument metadata.
48/// # Returns
49/// * `f64` representing the maximum available quantity.
50pub fn calculate_max_available_quantity(
51    cash: f64,
52    mut current_quantity: f64,
53    strategy_instruments_info: &instruments_info::InstrumentInfo,
54) -> f64 {
55    let margin = strategy_instruments_info.margin;
56    let precision = strategy_instruments_info.contract_precision as i32;
57    let max_quantity = ((cash / margin) * 10.0_f64.powi(precision)).floor() / 10.0_f64.powi(precision);
58
59    current_quantity = current_quantity.min(max_quantity.abs());
60
61    if current_quantity == 0.0 {
62        current_quantity += 1.0 / 10.0_f64.powi(precision);
63    }
64
65    current_quantity
66}

68/// Parses optimization configuration from strategy settings.
69/// # Arguments
70/// * `strategy_settings` - The strategy settings.
71/// # Returns
72/// * `OptimizationConfig` containing the parsed configuration.
73pub fn parse_optimization_config(
74    strategy_settings: &settings::StrategySettings,
75) -> optimization::OptimizationConfig {
76    let strategy_params_ranges: std::collections::HashMap<String, settings::ParamSpec> = strategy_settings
77        .strategy_params
78        .iter()
79        .map(|(name, spec)| (name.clone(), spec.clone()))
80        .collect();
81    
82    let pos_sizer_value_range = strategy_settings.pos_sizer_params.pos_sizer_value.clone();
83    let slippage_range = strategy_settings.slippage.clone();
84
85    let pos_sizer_additional_params: std::collections::HashMap<String, settings::ParamSpec> = strategy_settings
86        .pos_sizer_params
87        .pos_sizer_params
88        .iter()
89        .map(|(name, spec)| (name.clone(), spec.clone()))
90        .collect();
91
92    optimization::OptimizationConfig::new()
93        .with_strategy_params_ranges(strategy_params_ranges)
94        .with_pos_sizer_name(strategy_settings.pos_sizer_params.pos_sizer_name.clone())
95        .with_pos_sizer_value_ranges(pos_sizer_value_range)
96        .with_slippage_range(slippage_range)
97        .with_pos_sizer_additional_params(pos_sizer_additional_params)
98}

100/// Creates a new strategy settings object from a parameter set.
101/// # Arguments
102/// * `original_settings` - The original strategy settings.
103/// * `parameters` - The parameter set to use.
104/// # Returns
105/// * `StrategySettings` with updated parameters.
106pub fn create_stratagy_settings_from_params(
107    strategy_settings: &settings::StrategySettings,
108    params: &optimization::ParameterSet,
109) -> settings::StrategySettings {
110    let mut new_strategy_settings = strategy_settings.clone();
111
112    new_strategy_settings.pos_sizer_params.pos_sizer_value = settings::ParamSpec::Discrete(vec![serde_json::Value::Number(serde_json::Number::from_f64(*params.get_pos_sizer_value()).unwrap())]);
113    new_strategy_settings.slippage = settings::ParamSpec::Discrete(vec![serde_json::Value::Number(serde_json::Number::from_f64(*params.get_slippage()).unwrap())]);
114
115    let mut map = strategy_settings.strategy_params.clone();
116    for (key, selected_value) in params.get_strategy_params() {
117        map.insert(key.clone(), settings::ParamSpec::Discrete(vec![selected_value.clone()]));
118    }
119    new_strategy_settings.strategy_params = map;
120
121    let mut ps_params_map = strategy_settings.pos_sizer_params.pos_sizer_params.clone();
122    for (key, selected_value) in params.get_pos_sizer_additional_params() {
123        ps_params_map.insert(key.clone(), settings::ParamSpec::Discrete(vec![selected_value.clone()]));
124    }
125    new_strategy_settings.pos_sizer_params.pos_sizer_params = ps_params_map;
126
127    new_strategy_settings
128}

130/// Exports the equity curve to a CSV file.
131/// # Arguments
132/// * `filename` - The name of the output file.
133/// # Returns
134/// * `anyhow::Result<()>` indicating success or failure.

136    equity_series: &Vec<(chrono::DateTime<chrono::Utc>, f64)>,
137    strategy_settings: &settings::StrategySettings,
138) -> anyhow::Result<()> {

142    writeln!(file, "datetime;capital")?;
143    for point in equity_series {

145    }
146
147    anyhow::Ok(())
148}

158/// 
159/// # Arguments

161/// export_equity_drawdowns_to_csv(&drawdowns, &drawdowns_pct, &equity_series, &strategy_settings)?;
162/// Creates: {exit_results_path}/equity_series.csv
163pub fn export_equity_drawdowns_to_csv(
164    drawdowns: &Vec<f64>,
165    drawdowns_pct: &Vec<f64>,
166    equity_series: &[(chrono::DateTime<chrono::Utc>, f64)],
167    strategy_settings: &settings::StrategySettings,
168) -> anyhow::Result<()> {
169    let path = format!("{}/equity_series.csv", strategy_settings.exit_results_path);
170
171    let mut file = std::fs::File::create(path)?;
172    writeln!(file, "datetime;capital;drawdown;drawdown_pct")?;
173    for ((datetime_capital, drawdown), drawdown_pct) in 
174        equity_series.iter().zip(drawdowns.iter()).zip(drawdowns_pct.iter())
175    {
176        let (datetime, capital) = datetime_capital;
177        writeln!(
178            file,
179            "{};{};{};{}",
180            datetime.format("%Y-%m-%d %H:%M:%S"),
181            capital,
182            drawdown,
183            drawdown_pct
184        )?;
185    }
186
187    anyhow::Ok(())
188}

190/// Generates a vector of `size` `f64` values using Latin Hypercube Sampling (LHS) within a specified range.
191///
192/// This function divides the range `[min, max]` into `size` equal intervals and samples one value
193/// uniformly from each interval. If a `step` is provided, the sampled value is snapped to the
194/// nearest multiple of `step` within the `[min, max]` bounds. This ensures good coverage of the
195/// parameter space while allowing for discrete steps.
196///
197/// # Arguments
198/// * `min` - The lower bound of the range.
199/// * `max` - The upper bound of the range.
200/// * `step` - Optional step size to discretize the sampled values.
201/// * `size` - The number of values to generate.
202/// * `rng` - A mutable reference to a random number generator implementing `rand::Rng`.
203///
204/// # Returns
205/// A `Vec<f64>` containing the generated values, shuffled to randomize order.
206fn generate_lhs_values_f64(
207    min: f64,
208    max: f64,
209    step: Option<f64>,
210    size: usize,
211    mut rng: &mut impl rand::Rng
212) -> Vec<f64> {
213    let mut values = Vec::with_capacity(size);
214    // Calculate the width of each interval for LHS
215    let interval_size = (max - min) / size as f64;
216
217    // Sample one value from each interval
218    for i in 0..size {
219        let interval_start = min + (i as f64) * interval_size;
220        let interval_end = min + ((i + 1) as f64) * interval_size;
221
222        // Sample uniformly within the specific interval
223        let value = rand_distr::Uniform::new(interval_start, interval_end).sample(&mut rng);
224
225        // If a step is defined, snap the value to the nearest step increment within bounds
226        let processed_value = if let Some(step_val) = step {
227            let clamped_value = value.clamp(min, max);
228            let rounded_index = ((clamped_value - min) / step_val).round();
229            rounded_index * step_val + min
230        } else {
231            // Otherwise, use the sampled value directly
232            value
233        };
234
235        values.push(processed_value);
236    }
237
238    // Shuffle the resulting vector to remove any systematic ordering from interval sampling
239    values.shuffle(&mut rng);
240    values
241}

243/// Generates a vector of `size` `f64` values based on a `ParamSpec`.
244///
245/// This function handles both `ParamSpec::Range` and `ParamSpec::Discrete`.
246/// For `Range`, it delegates to `generate_lhs_values_f64` for Latin Hypercube Sampling.
247/// For `Discrete`, it cycles through the provided discrete values and shuffles the result.
248///
249/// # Arguments
250/// * `spec` - A reference to the `settings::ParamSpec` defining the parameter space.
251/// * `size` - The number of values to generate.
252/// * `rng` - A mutable reference to a random number generator implementing `rand::Rng`.
253///
254/// # Returns
255/// A `Vec<f64>` containing the generated values.
256fn generate_param_values(
257    spec: &settings::ParamSpec,
258    size: usize,
259    rng: &mut impl rand::Rng
260) -> Vec<f64> {
261    match spec {
262        // If the spec defines a continuous range, use LHS
263        settings::ParamSpec::Range { start, end, step } => {
264            generate_lhs_values_f64(*start, *end, Some(*step), size, rng)
265        }
266        // If the spec defines a discrete list of values
267        settings::ParamSpec::Discrete(values) => {
268            let mut result = Vec::with_capacity(size);
269            let num_discrete_vals = values.len();
270
271            // If the discrete list is empty, return an empty vector
272            if num_discrete_vals == 0 {
273                return vec![0.0; size];
274           }
275
276           // Create a list of indices, cycling through the discrete values to fill 'size' slots
277            let indices: Vec<usize> = (0..size).map(|i| i % num_discrete_vals).collect();
278            let mut shuffled_indices = indices;
279            // Shuffle the indices to randomize which discrete value goes where
280            shuffled_indices.shuffle(rng);
281
282            // Map the shuffled indices back to the actual discrete values
283            for idx in shuffled_indices {
284                if let Some(json_val) = values.get(idx) {
285                    // Attempt to extract an f64 value from the serde_json::Value
286                    if let Some(f64_val) = json_val.as_f64() {
287                        result.push(f64_val);
288                    } else {
289                        // Log a warning if the value isn't a number and push a default (0.0)
290                        eprintln!("Warning: Discrete value {:?} is not a number, skipping.", json_val);
291                        result.push(0.0);   // Consider returning Result or handling error differently
292                    }
293                }
294            }
295
296            result
297        }
298    }
299}

301/// Generates a vector of `ParameterSet` objects using Latin Hypercube Sampling (LHS) for specified parameter ranges.
302///
303/// This function creates a diverse initial population or fresh blood by sampling parameter values
304/// according to their specifications (either continuous `Range` or discrete `Discrete` lists).
305/// It ensures that the generated sets cover the defined parameter space effectively.
306///
307/// # Arguments
308/// * `strategy_params_ranges` - A map of strategy parameter names to their `ParamSpec`.
309/// * `pos_sizer_value_range` - The `ParamSpec` for the position sizer value.
310/// * `slippage_range` - The `ParamSpec` for the slippage value.
311/// * `pos_sizer_additional_params` - A map of additional position sizer parameter names to their `ParamSpec`.
312/// * `pos_sizer_name` - The name of the position sizing method.
313/// * `size` - The number of `ParameterSet` objects to generate.
314/// * `rng` - A mutable reference to a random number generator implementing `rand::Rng`.
315///
316/// # Returns
317/// A `Vec<optimization::ParameterSet>` containing the generated parameter sets.
318pub fn generate_lhs_parameter_sets(
319    strategy_params_ranges: &std::collections::HashMap<String, settings::ParamSpec>,
320    pos_sizer_value_range: &settings::ParamSpec,
321    slippage_range: &settings::ParamSpec,
322    pos_sizer_additional_params: &std::collections::HashMap<String, settings::ParamSpec>,
323    pos_sizer_name: &String,
324    size: usize,
325    rng: &mut impl rand::Rng
326) -> Vec<optimization::ParameterSet> {
327    let mut result = Vec::with_capacity(size);
328
329    // Generate LHS/dischcrete values for each strategy parameter across 'size' sets
330    let strategy_param_values: std::collections::HashMap<String, Vec<f64>> = strategy_params_ranges
331        .iter()
332        .map(|(name, spec)| {
333            let values = generate_param_values(spec, size, rng);
334            (name.clone(), values)
335        })
336        .collect();
337
338    // Generate LHS/discrete values for pos_sizer_value across 'size' sets
339    let pos_sizer_values = if pos_sizer_name == "1" {
340        vec![0.0; size]
341    } else {
342        generate_param_values(pos_sizer_value_range, size, rng)
343    };
344
345    // Generate LHS/discrete values for slippage across 'size' sets
346    let slippage_values = generate_param_values(slippage_range, size, rng);
347    // Generate LHS/discrete values for each additional pos sizer parameter across 'size' sets
348    let pos_sizer_additional_values: std::collections::HashMap<String, Vec<f64>> = pos_sizer_additional_params
349        .iter()
350        .map(|(name, spec)| {
351            let values = generate_param_values(spec, size, rng);
352            (name.clone(), values)
353        })
354        .collect();
355
356    // Construct 'size' ParameterSet objects by combining values from each parameter map
357    for i in 0..size {
358        // Build the strategy_params vector for the i-th ParameterSet
359        let strategy_params: Vec<(String, serde_json::Value)> = strategy_param_values
360            .iter()
361            .map(|(name, values)| {
362                // Get the i-th value for this parameter, defaulting to 0.0 if index is somehow out of bounds
363                let value = values.get(i).copied().unwrap_or(0.0);
364                (name.clone(), serde_json::Value::Number(serde_json::Number::from_f64(value).unwrap()))
365            })
366            .collect();
367
368        // Build the pos_sizer_additional_params vector for the i-th ParameterSet
369        let pos_sizer_additional_params: Vec<(String, serde_json::Value)> = pos_sizer_additional_values
370        .iter()
371        .map(|(name, values)| {
372            // Get the i-th value for this parameter, defaulting to 0.0 if index is out of bounds
373            let value = values.get(i).copied().unwrap_or(0.0);
374            (name.clone(), serde_json::Value::Number(serde_json::Number::from_f64(value).unwrap()))
375        })
376        .collect();
377
378        // Create the ParameterSet for index i
379        let param_set = optimization::ParameterSet::new()
380            .with_strategy_params(strategy_params)
381            .with_pos_sizer_name(pos_sizer_name.clone())
382            .with_pos_sizer_additional_params(pos_sizer_additional_params)
383            .with_pos_sizer_value(pos_sizer_values[i])
384            .with_slippage(slippage_values[i]);
385
386        result.push(param_set);
387    }
388
389    result
390}

392/// Helper function to extract a parameter value as usize from the strategy settings.
393/// It checks for the parameter in the map, verifies it's a number, and converts it to usize.
394///
395/// # Arguments
396/// * `params` - The map of strategy parameters.
397/// * `name` - The name of the parameter to extract.
398///
399/// # Returns
400/// * `anyhow::Result<usize>` - The parameter value as usize or an error.
401pub fn get_param_as_usize(params: &std::collections::HashMap<String, settings::ParamSpec>, name: &str) -> anyhow::Result<usize> {
402    let param_spec = params
403        .get(name)
404        .ok_or_else(|| anyhow::anyhow!("Missing parameter '{}'", name))?;
405    
406    let value = match param_spec {
407        settings::ParamSpec::Discrete(values) => {
408            if let Some(v) = values.first() {
409                v
410            } else {
411                anyhow::bail!("Parameter '{}' has no values in Discrete list", name);
412            }
413        }
414        settings::ParamSpec::Range { .. } => {
415            anyhow::bail!("Parameter '{}' is a Range, but a single value is expected", name);
416        }
417    };
418
419    if let Some(val) = value.as_u64() {
420        anyhow::Ok(val as usize)
421    } else if let Some(val) = value.as_f64() {
422        anyhow::Ok(val as usize)
423    } else {
424        Err(anyhow::anyhow!("Parameter '{}' must be a number, got: {:?}", name, value))
425    }
426}
