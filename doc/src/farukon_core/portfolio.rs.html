3//! Portfolio manager: tracks positions, holdings, equity, and risk.
4//! Implements PortfolioHandler trait for integration with Backtest.
5//! Handles fill events, signal events, and margin calls.
6
7use crate::event;

11/// Represents the state of a position for a specific symbol.
12/// Tracks deal count, size, entry/exit prices, and capital involved.
13#[derive(Debug, Clone)]
14pub struct PositionState {
15    /// Number of deals executed for this symbol.
16    pub deal_number: usize,
17    /// Current size of the position (positive for long, negative for short).
18    pub position: f64,
19    /// Capital allocated when entering the current position.
20    pub entry_capital: f64,
21    /// Price at which the current position was opened.
22    pub entry_price: Option<f64>,
23}
24
25impl PositionState {
26    /// Creates a new `PositionState` with default values (no position, no deals).
27    pub fn new() -> Self {
28        Self {
29            deal_number: 0,
30            position: 0.0,
31            entry_capital: 0.0,
32            entry_price: None,
33        }
34    }
35
36}
37
38/// A snapshot of all position states at a specific point in time.
39#[derive(Debug, Clone)]
40pub struct PositionSnapshot {
41    pub datetime: chrono::DateTime<chrono::Utc>,
42    pub positions: std::collections::HashMap<String, PositionState>,
43}
44
45impl PositionSnapshot {
46    /// Creates a new `PositionSnapshot`.
47    ///
48    /// # Arguments
49    /// * `datetime` - The timestamp for the snapshot.
50    /// * `positions` - The map of symbol states.
51    pub fn new(
52        datetime: chrono::DateTime<chrono::Utc>,
53        positions: std::collections::HashMap<String, PositionState>,
54    ) -> Self {
55        Self { datetime, positions }
56    }
57
58}

60/// Represents the state of holdings (PnL, blocked margin) for a specific symbol.
61#[derive(Debug, Clone)]
62pub struct HoldingsState {
63    /// Profit and Loss for this symbol.
64    pub pnl: f64,
65    /// Margin blocked by open positions for this symbol.
66    pub blocked: f64,
67    pub signal_name: Option<String>,
68}
69
70impl HoldingsState {
71    /// Creates a new `HoldingsState` with default values (zero PnL, zero blocked margin).
72    pub fn new() -> Self {
73        Self { pnl: 0.0, blocked: 0.0, signal_name: None }
74    }
75
76}
77
78/// A snapshot of all holding states at a specific point in time.
79#[derive(Debug, Clone)]
80pub struct HoldingSnapshot {
81    /// The timestamp of this snapshot.
82    pub datetime: chrono::DateTime<chrono::Utc>,
83    /// Total capital (cash + realized PnL + unrealized PnL).
84    pub capital: f64,
85    /// Available cash (not blocked by positions).
86    pub cash: f64,
87    /// Total capital blocked by all open positions.
88    pub blocked: f64,
89    /// A map of symbol names to their `HoldingsState`.
90    pub holdings: std::collections::HashMap<String, HoldingsState>,
91}
92
93impl HoldingSnapshot {
94    /// Creates a new `HoldingSnapshot`.
95    ///
96    /// # Arguments
97    /// * `datetime` - The timestamp for the snapshot.
98    /// * `holdings` - The map of symbol states.
99    pub fn new(
100        datetime: chrono::DateTime<chrono::Utc>,
101        capital: f64,
102        cash: f64,
103        blocked: f64,
104        holdings: std::collections::HashMap<String, HoldingsState>,
105    ) -> Self {
106        Self { datetime, capital, cash, blocked, holdings }
107    }
108
109}

111/// Defines the interface for a portfolio manager.
112/// Allows the backtesting engine to interact with the portfolio state without knowing its concrete implementation.
113pub trait PortfolioHandler {
114    /// Updates the portfolio's time-indexed state (e.g., equity, holdings, positions) based on the latest market data.
115    /// Called on every `MARKET` event.
116    fn update_timeindex(&mut self, data_handler: &Box<dyn data_handler::DataHandler>);
117    
118    /// Processes a `SIGNAL` event generated by a strategy.
119    /// Typically converts the signal into an `ORDER` event.
120    fn update_signal(&mut self, signal_event: &event::SignalEvent, data_handler: &Box<dyn data_handler::DataHandler>);
121    
122    /// Processes a `FILL` event received from the execution handler.
123    /// Updates positions and holdings based on the executed trade details.
124    fn update_fill(
125        &mut self,
126        fill_event: &event::FillEvent,
127        data_handler: &Box<dyn data_handler::DataHandler>,
128    );
129
130    /// Updates the internal position state based on a `FILL` event.
131    fn update_positions_from_fill(&mut self, fill_event: &event::FillEvent, data_handler: &Box<dyn data_handler::DataHandler>);
132    
133    /// Updates the internal holding state (PnL, blocked margin) based on a `FILL` event.
134    fn update_holdings_from_fill(
135        &mut self,
136        fill_event: &event::FillEvent,
137        data_handler: &Box<dyn data_handler::DataHandler>,
138    );
139
140    // --- Getters ---
141
142    fn get_current_positions(&self) -> &std::collections::HashMap<String, PositionState>;
143    fn get_all_positions(&self) -> &Vec<PositionSnapshot>;
144    fn get_current_holdings(&self) -> &std::collections::HashMap<String, HoldingsState>;
145    fn get_all_holdings(&self) -> &Vec<HoldingSnapshot>;
146    fn get_latest_holdings(&self) -> Option<&HoldingSnapshot>;
147    fn get_equity_capital_values(&self) -> Vec<f64>;
148    fn output_summary_stats(&self) -> anyhow::Result<&performance::PerformanceMetrics>;
149    fn calculate_final_performance(&mut self);
150
151}
