3//! Technical indicator library.
4//!
5//! This module provides foundational technical indicators used by trading strategies
6//! within the Farukon event-driven backtesting framework. Indicators operate on
7//! time-series data (e.g., OHLCV) supplied via iterators, enabling decoupling from
8//! the underlying data storage mechanism.
9//!
10//! Currently implemented indicators:
11//! - Simple Moving Average (`sma`)
12//! - Highest High over N periods (`highest`)
13//! - Lowest Low over N periods (`lowest`)
14//!
15//! All functions are pure, stateless, and return `None` when insufficient historical
16//! data is available for the requested lookback window.
17
18/// Computes the **Simple Moving Average (SMA)** over the last `n` data points.
19///
20/// The SMA is the unweighted mean of the previous `n` values. It is commonly used
21/// to smooth price data and identify trend direction.
22///
23/// # Type Parameters
24/// * `I` — An iterator yielding references to `f64` values (e.g., closing prices).
25///
26/// # Arguments
27/// * `dh` — An iterator over historical data points (e.g., a slice of `&f64`).
28/// * `n` — The lookback period (number of bars to average). Must be ≥ 1.
29///
30/// # Returns
31/// * `Some(f64)` — The computed SMA value.
32/// * `None` — If `n == 0` or fewer than `n` data points are available.
33pub fn sma<'a, I>(
34    dh: I,
35    n: usize,
36) -> Option<f64>
37where
38    I: IntoIterator<Item = &'a f64>,
39    I::IntoIter: Clone,
40{
41    // Simple Moving Average.
42    // Returns average of last n bars.
43    if n == 0 { return None; }
44
45    // Convert to Vec
46    let data: Vec<f64> = dh.into_iter().copied().collect();
47    let total_count = data.len();
48
49    // If insufficient data is available, return None.
50    if total_count < n { return None; }
51
52    // Calculate the sum of the last `n` bars, starting from `shift` bars ago.
53    let start = total_count - n;
54    let end = start + n;
55    let sum: f64 = data[start..end]
56        .iter()
57        .sum();
58    // Return the average.
59    Some(sum / (n as f64))
60}

62/// Finds the **highest value** over the last `n` bars, shifted back by `shift` bars.
63///
64/// This function is typically used with high prices to identify resistance levels
65/// or channel boundaries. It skips any `None` values in the input sequence.
66///
67/// # Type Parameters
68/// * `I` — An iterator yielding references to `Option<f64>` (e.g., high prices that may be missing).
69///
70/// # Arguments
71/// * `dh` — An iterator over optional historical data points.
72/// * `n` — The lookback period (number of bars to scan). Must be ≥ 1.
73/// * `shift` — Number of bars to shift the window backward (0 = current bar included).
74///
75/// # Returns
76/// * `Some(f64)` — The maximum value in the specified window.
77/// * `None` — If `n == 0`, `n + shift == 0`, or insufficient data is available.
78///
79/// # Note
80/// The effective required data length is `n + shift`. For example, to get the highest
81/// high of the last 5 bars as of 2 bars ago, use `n=5, shift=2` (requires 7 bars total).
82pub fn highest<'a, I>(
83    dh: I,
84    n: usize,
85    shift: usize,
86) -> Option<f64> 
87where
88    I: IntoIterator<Item = &'a Option<f64>>,
89    I::IntoIter: Clone,
90{
91    if n == 0 { return None; }
92
93    let need = n + shift;
94    if need == 0 { return None; }
95    
96    // Convert to Vec, filtering out None values
97    let data: Vec<Option<f64>> = dh
98        .into_iter()
99        .copied()
100        .collect();
101    let total_count = data.len();
102
103    // If insufficient data is available, return None.
104    if total_count < need { return None; }
105    
106    let start = total_count - need;
107    let end = start + n;
108
109    let window = &data[start..end];
110    for item in window {
111        if item.is_none() {
112            return None;
113        }
114    }
115
116    window
117        .iter()
118        .filter_map(|&x| x)
119        .max_by(|a, b| a.partial_cmp(b).unwrap_or(std::cmp::Ordering::Equal))
120}

144    n: usize,

154    if need == 0 { return None; }
