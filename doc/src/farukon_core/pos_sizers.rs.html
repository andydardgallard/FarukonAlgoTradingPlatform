3//! Position sizing module.
4//!
5//! This module implements risk-aware position sizing strategies that determine trade quantity
6//! based on available capital, instrument metadata, and user-defined risk parameters.
7//! It supports multiple sizing methods, including fixed-lot and risk-based approaches,
8//! and incorporates transaction costs (commissions) where applicable.
9//!
10//! Currently supported position sizers:
11//! - `"1"`: Fixed-size lot (default: 1 contract).
12//! - `"mpr"`: Maximum Percent Risk — sizes positions so that the potential loss
13//!            (including commissions) does not exceed a user-defined percentage of capital.
14//! - `"poe"`: (Planned) Percent of Equity — sizes positions as a percentage of total equity.
15//!
16//! Commission handling is exchange-specific. As of now, only FORTS exchange commissions
17//! are modeled, using the `calculate_forts_comission` function from the `commission_plans` module.
18//! Commissions for both entry and exit are included in risk calculations for the MPR method.
19
20use crate::settings;
21use crate::commission_plans;
22use crate::instruments_info;

24/// Calculates the position size using the "MPR" (Maximum Possible Risk) method.
25/// # Arguments
26/// * `mode` - Operational mode (Debug, Optimize, etc.).
27/// * `capital` - Available capital.
28/// * `entry_price` - Entry price for the trade.
29/// * `exit_price` - Exit price for the trade.
30/// * `strategy_settings` - Strategy settings.
31/// * `instrument_info` - Instrument metadata.
32/// # Returns
33/// * An optional `f64` representing the quantity to trade.
34fn mpr(
35    mode: &String,
36    capital: f64,
37    entry_price: f64,
38    exit_price: f64,
39    strategy_settings: &settings::StrategySettings,
40    strategy_instruments_info_for_symbol: &instruments_info::InstrumentInfo,
41) -> Option<f64> {
42    let mpr = {
43        match &strategy_settings.pos_sizer_params.pos_sizer_value {
44            settings::ParamSpec::Discrete(values) => {
45                if let Some(value) = values.first() {
46                    value.as_f64()?
47                } else {
48                    return None;
49                }
50            }
51            settings::ParamSpec::Range { .. } => {
52                return None;
53            }
54        }
55    };
56
57    let full_commission = match strategy_instruments_info_for_symbol.exchange.as_str() {
58            "FORTS" => {
59                let entry_commission = commission_plans::calculate_forts_comission(
60                    Some(entry_price),
61                    strategy_instruments_info_for_symbol,
62                    strategy_settings,
63                ).unwrap();
64                let exit_commission = commission_plans::calculate_forts_comission(
65                    Some(exit_price),
66                    strategy_instruments_info_for_symbol,
67                    strategy_settings
68                ).unwrap();
69
70                Some(entry_commission + exit_commission)
71            } 
72            _ => None,
73    };
74
75    let risk_per_deal_in_points = (exit_price - entry_price).abs();
76    let point_value = ((strategy_instruments_info_for_symbol.step_price / strategy_instruments_info_for_symbol.step) * 100_000.0).round() / 100_000.0;
77    let risk_per_deal_in_value_net = risk_per_deal_in_points * point_value;
78    let risk_per_deal_in_value_gross = risk_per_deal_in_value_net + full_commission?;
79    let max_percent_risk = capital * (mpr / 100.0);
80    let points_from_zero = strategy_instruments_info_for_symbol.contract_precision as i32;
81
82    if mode == "Debug" {
83        println!(
84            "risk_per_deal_in_points: {}, point_value: {}, risk_per_deal_in_value_net: {}, risk_per_deal_in_value_gross: {}, max_percent_risk: {}, capital: {}",
85            risk_per_deal_in_points, point_value, risk_per_deal_in_value_net, risk_per_deal_in_value_gross, max_percent_risk, capital
86        );
87    }
88                    
89    Some(((max_percent_risk / risk_per_deal_in_value_gross) * 10.0_f64.powi(points_from_zero)).floor() / 10.0_f64.powi(points_from_zero))
90}

92/// Returns a fixed position size of 1.0 contract.
93///
94/// Primarily used for baseline testing or strategies that do not require dynamic sizing.
95/// In `"Debug"` mode, logs point value and entry commission for diagnostics.
96fn plain_pos_sizer(
97    mode: &String,
98    entry_price: f64,
99    strategy_settings: &settings::StrategySettings,
100    strategy_instruments_info_for_symbol: &instruments_info::InstrumentInfo,
101) -> Option<f64> {
102    if mode == "Debug" {
103        let point_value = ((strategy_instruments_info_for_symbol.step_price / strategy_instruments_info_for_symbol.step) * 100_000.0).round() / 100_000.0;
104        let commission = match strategy_instruments_info_for_symbol.exchange.as_str() {
105            "FORTS" => {
106                let entry_commission = commission_plans::calculate_forts_comission(
107                    Some(entry_price),
108                    strategy_instruments_info_for_symbol,
109                    strategy_settings,
110                ).unwrap();
111
112                Some(entry_commission)
113            } 
114            _ => None,
115        };
116
117        println!("point_value: {}, commision: {:?}", point_value, commission);
118    }
119    Some(1.0)
120}

122/// Calculates the position size based on the strategy settings.
123/// # Arguments
124/// * `mode` - Operational mode.
125/// * `capital` - Available capital.
126/// * `price` - Current price.
127/// * `long_sma` - Long-term SMA (optional).
128/// * `strategy_settings` - Strategy settings.
129/// * `instrument_info` - Instrument metadata.
130/// # Returns
131/// * An optional `f64` representing the quantity to trade.
132pub fn get_pos_sizer_from_settings(
133    mode: &String,
134    capital: Option<f64>,
135    entry_price: Option<f64>,
136    exit_price: Option<f64>,
137    strategy_settings: &settings::StrategySettings,
138    strategy_instruments_info_for_symbol: &instruments_info::InstrumentInfo,
139) -> Option<f64> {
140    let is_single_value = match &strategy_settings.pos_sizer_params.pos_sizer_value {
141        settings::ParamSpec::Discrete(values) => values.len() == 1,
142        settings::ParamSpec::Range { .. } => false,
143    };
144
145    if is_single_value {
146        let quantity = match strategy_settings.pos_sizer_params.pos_sizer_name.as_str() {
147            "1" => plain_pos_sizer(
148                mode,
149                entry_price?,
150                strategy_settings,
151                strategy_instruments_info_for_symbol
152            ),
153            "mpr" => mpr(
154                mode,
155                capital?,
156                entry_price?,
157                exit_price?,
158                strategy_settings,
159                strategy_instruments_info_for_symbol,
160            ),
161            "poe" => None, // TODO
162            _ => None,
163        };
164        
165        quantity
166    } else { None }
167}
