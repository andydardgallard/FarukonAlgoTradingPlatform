4    kCFAllocatorDefault, CFArray, CFDictionary, CFNumber, CFRetained, CFString,

6use objc2_io_kit::{IOHIDEventSystemClient, IOHIDServiceClient};

8use crate::sys::inner::ffi::{
9    kHIDPage_AppleVendor, kHIDUsage_AppleVendor_TemperatureSensor, kIOHIDEventTypeTemperature,
10    IOHIDEventFieldBase, IOHIDEventGetFloatValue, IOHIDEventSystemClientCreate,
11    IOHIDEventSystemClientSetMatching, IOHIDServiceClientCopyEvent,

17pub(crate) struct ComponentsInner {
18    pub(crate) components: Vec<Component>,
19    client: Option<CFRetained<IOHIDEventSystemClient>>,
20}
21
22// SAFETY: `ComponentsInner::client` is never updated in a `&self` context, so it's safe to
23// make the type `Sync` and `Send`.
24unsafe impl Send for ComponentsInner {}
25unsafe impl Sync for ComponentsInner {}
26
27impl ComponentsInner {
28    pub(crate) fn new() -> Self {
29        Self {
30            components: vec![],
31            client: None,
32        }
33    }
34
35    pub(crate) fn from_vec(components: Vec<Component>) -> Self {
36        Self {
37            components,
38            client: None,
39        }
40    }
41
42    pub(crate) fn into_vec(self) -> Vec<Component> {
43        self.components
44    }
45
46    pub(crate) fn list(&self) -> &[Component] {
47        &self.components
48    }
49
50    pub(crate) fn list_mut(&mut self) -> &mut [Component] {
51        &mut self.components
52    }
53
54    #[allow(unreachable_code)]
55    pub(crate) fn refresh(&mut self) {
56        let keys = [
57            &*CFString::from_static_str(HID_DEVICE_PROPERTY_PRIMARY_USAGE_PAGE),
58            &*CFString::from_static_str(HID_DEVICE_PROPERTY_PRIMARY_USAGE),
59        ];
60
61        let nums = [
62            &*CFNumber::new_i32(kHIDPage_AppleVendor),
63            &*CFNumber::new_i32(kHIDUsage_AppleVendor_TemperatureSensor),
64        ];
65
66        let matches = CFDictionary::from_slices(&keys, &nums);
67        let matches = matches.as_opaque();
68
69        unsafe {
70            if self.client.is_none() {
71                let client = match IOHIDEventSystemClientCreate(kCFAllocatorDefault) {
72                    // SAFETY: `IOHIDEventSystemClientCreate` is a "create"
73                    // function, so the client has +1 retain count.
74                    Some(c) => CFRetained::from_raw(c),
75                    None => return,
76                };
77                self.client = Some(client);
78            }
79
80            let client = self.client.as_ref().unwrap();
81
82            let _ = IOHIDEventSystemClientSetMatching(client, matches);
83
84            let services = match client.services() {
85                Some(s) => s,
86                None => return,
87            };
88
89            // SAFETY: Return type documented to be CFArray of IOHIDServiceClient.
90            let services = CFRetained::cast_unchecked::<CFArray<IOHIDServiceClient>>(services);
91
92            let key = CFString::from_static_str(HID_DEVICE_PROPERTY_PRODUCT);
93
94            for service in services {
95                let Some(name) = service.property(&key) else {
96                    continue;
97                };
98                let name = name.downcast::<CFString>().unwrap();
99                let name_str = name.to_string();
100
101                if let Some(c) = self
102                    .components
103                    .iter_mut()
104                    .find(|c| c.inner.label == name_str)
105                {
106                    c.refresh();
107                    c.inner.updated = true;
108                    continue;
109                }
110
111                let mut component = ComponentInner::new(name_str, None, None, service);
112                component.refresh();
113
114                self.components.push(Component { inner: component });
115            }
116        }
117    }
118}

120pub(crate) struct ComponentInner {
121    service: CFRetained<IOHIDServiceClient>,
122    temperature: Option<f32>,
123    label: String,
124    max: f32,
125    critical: Option<f32>,
126    pub(crate) updated: bool,
127}
128
129// SAFETY: `ComponentsInner::service` is never updated, so it's safe to make the type `Sync`
130// and `Send`.
131unsafe impl Send for ComponentInner {}
132unsafe impl Sync for ComponentInner {}
133
134impl ComponentInner {
135    pub(crate) fn new(
136        label: String,
137        max: Option<f32>,
138        critical: Option<f32>,
139        service: CFRetained<IOHIDServiceClient>,
140    ) -> Self {
141        Self {
142            service,
143            label,
144            max: max.unwrap_or(0.),
145            critical,
146            temperature: None,
147            updated: true,
148        }
149    }
150
151    pub(crate) fn temperature(&self) -> Option<f32> {
152        self.temperature
153    }
154
155    pub(crate) fn max(&self) -> Option<f32> {
156        Some(self.max)
157    }
158
159    pub(crate) fn critical(&self) -> Option<f32> {
160        self.critical
161    }
162
163    pub(crate) fn label(&self) -> &str {
164        &self.label
165    }
166
167    pub(crate) fn refresh(&mut self) {
168        unsafe {
169            let Some(event) =
170                IOHIDServiceClientCopyEvent(&self.service, kIOHIDEventTypeTemperature, 0, 0)
171            else {
172                self.temperature = None;
173                return;
174            };
175            // SAFETY: `IOHIDServiceClientCopyEvent` is a "copy" function, so
176            // the event has +1 retain count.
177            let event = CFRetained::from_raw(event);
178
179            let temperature =
180                IOHIDEventGetFloatValue(&event, IOHIDEventFieldBase(kIOHIDEventTypeTemperature))
181                    as _;
182            self.temperature = Some(temperature);
183            if temperature > self.max {
184                self.max = temperature;
185            }
186        }
187    }
188}
