6use libc::{
7    host_processor_info, mach_port_t, munmap, natural_t, processor_cpu_load_info,
8    processor_cpu_load_info_t, sysconf, vm_page_size, PROCESSOR_CPU_LOAD_INFO, _SC_CLK_TCK,

10use std::ptr::null_mut;

13struct ProcessorCpuLoadInfo {
14    cpu_load: processor_cpu_load_info_t,
15    cpu_count: natural_t,
16}
17
18impl ProcessorCpuLoadInfo {
19    fn new(port: mach_port_t) -> Option<Self> {
20        let mut info_size = std::mem::size_of::<processor_cpu_load_info_t>() as _;
21        let mut cpu_count = 0;
22        let mut cpu_load: processor_cpu_load_info_t = null_mut();
23
24        unsafe {
25            if host_processor_info(
26                port,
27                PROCESSOR_CPU_LOAD_INFO,
28                &mut cpu_count,
29                &mut cpu_load as *mut _ as *mut _,
30                &mut info_size,
31            ) != 0
32            {
33                sysinfo_debug!("host_processor_info failed, not updating CPU ticks usage...");
34                None
35            } else if cpu_count < 1 || cpu_load.is_null() {
36                None
37            } else {
38                Some(Self {
39                    cpu_load,
40                    cpu_count,
41                })
42            }
43        }
44    }
45}
46
47impl Drop for ProcessorCpuLoadInfo {
48    fn drop(&mut self) {
49        unsafe {
50            munmap(self.cpu_load as _, vm_page_size);
51        }
52    }
53}

55pub(crate) struct SystemTimeInfo {
56    timebase_to_ns: f64,
57    pub(crate) timebase_to_ms: f64,
58    clock_per_sec: f64,
59    old_cpu_info: ProcessorCpuLoadInfo,
60    last_update: Option<Instant>,
61    prev_time_interval: f64,
62}
63
64unsafe impl Send for SystemTimeInfo {}
65unsafe impl Sync for SystemTimeInfo {}
66
67impl SystemTimeInfo {
68    #[allow(deprecated)] // Everything related to mach_timebase_info_data_t
69    pub fn new(port: mach_port_t) -> Option<Self> {
70        unsafe {
71            let clock_ticks_per_sec = sysconf(_SC_CLK_TCK);
72
73            // FIXME: Maybe check errno here? Problem is that if errno is not 0 before this call,
74            //        we will get an error which isn't related...
75            // if let Some(er) = std::io::Error::last_os_error().raw_os_error() {
76            //     if err != 0 {
77            //         println!("==> {:?}", er);
78            //         sysinfo_debug!("Failed to get _SC_CLK_TCK value, using old CPU tick measure system");
79            //         return None;
80            //     }
81            // }
82
83            let mut info = mach_timebase_info_data_t { numer: 0, denom: 0 };
84            if mach_timebase_info(&mut info) != libc::KERN_SUCCESS {
85                sysinfo_debug!("mach_timebase_info failed, using default value of 1");
86                info.numer = 1;
87                info.denom = 1;
88            }
89
90            let old_cpu_info = match ProcessorCpuLoadInfo::new(port) {
91                Some(cpu_info) => cpu_info,
92                None => {
93                    sysinfo_debug!("host_processor_info failed, using old CPU tick measure system");
94                    return None;
95                }
96            };
97
98            let nano_per_seconds = 1_000_000_000.;
99            let timebase_to_ns = info.numer as f64 / info.denom as f64;
100            sysinfo_debug!("");
101            Some(Self {
102                timebase_to_ns,
103                // We convert from nano (10^-9) to ms (10^3).
104                timebase_to_ms: timebase_to_ns / 1_000_000.,
105                clock_per_sec: nano_per_seconds / clock_ticks_per_sec as f64,
106                old_cpu_info,
107                last_update: None,
108                prev_time_interval: 0.,
109            })
110        }
111    }
112
113    pub fn get_time_interval(&mut self, port: mach_port_t) -> f64 {
114        let need_cpu_usage_update = self
115            .last_update
116            .map(|last_update| last_update.elapsed() >= crate::MINIMUM_CPU_UPDATE_INTERVAL)
117            .unwrap_or(true);
118        if need_cpu_usage_update {
119            let mut total = 0;
120            let new_cpu_info = match ProcessorCpuLoadInfo::new(port) {
121                Some(cpu_info) => cpu_info,
122                None => return 0.,
123            };
124            let cpu_count = std::cmp::min(self.old_cpu_info.cpu_count, new_cpu_info.cpu_count);
125            unsafe {
126                for i in 0..cpu_count {
127                    let new_load: &processor_cpu_load_info = &*new_cpu_info.cpu_load.offset(i as _);
128                    let old_load: &processor_cpu_load_info =
129                        &*self.old_cpu_info.cpu_load.offset(i as _);
130                    for (new, old) in new_load.cpu_ticks.iter().zip(old_load.cpu_ticks.iter()) {
131                        if new > old {
132                            total += new.saturating_sub(*old);
133                        }
134                    }
135                }
136            }
137
138            self.old_cpu_info = new_cpu_info;
139            self.last_update = Some(Instant::now());
140
141            // Now we convert the ticks to nanoseconds (if the interval is less than
142            // `MINIMUM_CPU_UPDATE_INTERVAL`, we replace it with it instead):
143            let base_interval = total as f64 / cpu_count as f64 * self.clock_per_sec;
144            let smallest = crate::MINIMUM_CPU_UPDATE_INTERVAL.as_secs_f64() * 1_000_000_000.0;
145            self.prev_time_interval = if base_interval < smallest {
146                smallest
147            } else {
148                base_interval / self.timebase_to_ns
149            };
150            self.prev_time_interval
151        } else {
152            self.prev_time_interval
153        }
154    }
155}

157// Get a property value from the IOPlatformExpertDevice.
158#[cfg(not(feature = "apple-sandbox"))]
159pub(crate) fn get_io_platform_property(key: &str) -> Option<String> {
160    use crate::sys::macos::utils::IOReleaser;
161    use objc2_core_foundation::{kCFAllocatorDefault, CFData, CFGetTypeID, CFString, ConcreteType};
162    use objc2_io_kit::{
163        kIOMasterPortDefault, IORegistryEntryCreateCFProperty, IOServiceGetMatchingService,
164        IOServiceMatching,
165    };
166    use std::ffi::CStr;
167
168    let matching = match unsafe { IOServiceMatching(b"IOPlatformExpertDevice\0".as_ptr().cast()) } {
169        Some(matching) => matching,
170        None => {
171            sysinfo_debug!("IOServiceMatching call failed, `IOPlatformExpertDevice` not found");
172            return None;
173        }
174    };
175
176    let result = unsafe {
177        IOServiceGetMatchingService(kIOMasterPortDefault, Some(matching.as_opaque().into()))
178    };
179    if result == 0 {
180        sysinfo_debug!("IOServiceGetMatchingService failed");
181        return None;
182    }
183
184    let _r = IOReleaser::new(result);
185    let key_cfstring = CFString::from_str(key);
186    let properties = unsafe {
187        IORegistryEntryCreateCFProperty(result, Some(&key_cfstring), kCFAllocatorDefault, 0)
188    }?;
189
190    if CFGetTypeID(Some(&*properties)) == CFString::type_id() {
191        properties
192            .downcast::<CFString>()
193            .ok()
194            .map(|s| s.to_string())
195    } else {
196        properties.downcast::<CFData>().ok().and_then(|s| {
197            CStr::from_bytes_with_nul(&s.to_vec())
198                .ok()
199                .and_then(|cstr| cstr.to_str().ok())
200                .map(str::to_owned)
201        })
202    }
203}

205#[cfg(test)]
206mod test {

208    use super::*;
209
210    /// Regression test for <https://github.com/GuillaumeGomez/sysinfo/issues/956>.
211    #[test]
212    fn test_getting_time_interval() {
213        if !crate::IS_SUPPORTED_SYSTEM || cfg!(feature = "apple-sandbox") {
214            return;
215        }
216
217        #[allow(deprecated)]
218        let port = unsafe { libc::mach_host_self() };
219        let mut info = SystemTimeInfo::new(port).unwrap();
220        info.get_time_interval(port);
221
222        std::thread::sleep(crate::MINIMUM_CPU_UPDATE_INTERVAL.saturating_mul(5));
223
224        let val = info.get_time_interval(port);
225        assert_ne!(
226            val,
227            crate::MINIMUM_CPU_UPDATE_INTERVAL.as_secs_f64() * 1_000_000_000.0
228        );
229    }
