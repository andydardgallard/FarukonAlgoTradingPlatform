79    pub(crate) fn refresh_specifics(&mut self, refresh_kind: DiskRefreshKind) -> bool {
80        self.refresh_kind(refresh_kind);
81        self.refresh_io(refresh_kind);
82
83        if refresh_kind.storage() {
84            unsafe {
85                if let Some(requested_properties) = build_requested_properties(&[
86                    kCFURLVolumeTotalCapacityKey,
87                    kCFURLVolumeAvailableCapacityKey,
88                    kCFURLVolumeAvailableCapacityForImportantUsageKey,
89                ]) {
90                    match get_disk_properties(&self.volume_url, &requested_properties) {
91                        Some(disk_props) => {
92                            match get_int_value(&disk_props, kCFURLVolumeTotalCapacityKey) {
93                                Some(total_space) => self.total_space = total_space,
94                                None => {
95                                    sysinfo_debug!("Failed to get disk total space");
96                                }
97                            }
98                            match get_available_volume_space(&disk_props) {
99                                Some(available_space) => self.available_space = available_space,
100                                None => {
101                                    sysinfo_debug!("Failed to get disk available space");
102                                }
103                            }
104                        }
105                        None => {
106                            sysinfo_debug!("Failed to get disk properties");
107                        }
108                    }
109                } else {
110                    sysinfo_debug!("failed to create volume key list, skipping refresh");
111                }
112            }
113        }
114
115        true
116    }

118    pub(crate) fn usage(&self) -> DiskUsage {
119        DiskUsage {
120            read_bytes: self.read_bytes.saturating_sub(self.old_read_bytes),
121            total_read_bytes: self.read_bytes,
122            written_bytes: self.written_bytes.saturating_sub(self.old_written_bytes),
123            total_written_bytes: self.written_bytes,
124        }
125    }
126
127    fn refresh_kind(&mut self, refresh_kind: DiskRefreshKind) {
128        if refresh_kind.kind() && self.type_ == DiskKind::Unknown(-1) {
129            #[cfg(target_os = "macos")]
130            {
131                match self
132                    .bsd_name
133                    .as_ref()
134                    .and_then(|name| crate::sys::inner::disk::get_disk_type(name))
135                {
136                    Some(type_) => self.type_ = type_,
137                    None => {
138                        sysinfo_debug!("Failed to retrieve `DiskKind`");
139                    }
140                }
141            }
142            #[cfg(not(target_os = "macos"))]
143            {
144                self.type_ = DiskKind::SSD;
145            }
146        }
147    }
148
149    #[cfg(target_os = "macos")]
150    fn refresh_io(&mut self, refresh_kind: DiskRefreshKind) {
151        if refresh_kind.io_usage() {
152            match self
153                .bsd_name
154                .as_ref()
155                .and_then(|name| crate::sys::inner::disk::get_disk_io(name))
156            {
157                Some((read_bytes, written_bytes)) => {
158                    self.old_read_bytes = self.read_bytes;
159                    self.old_written_bytes = self.written_bytes;
160                    self.read_bytes = read_bytes;
161                    self.written_bytes = written_bytes;
162                }
163                None => {
164                    sysinfo_debug!("Failed to update disk i/o stats");
165                }
166            }
167        }
168    }
169
170    #[cfg(not(target_os = "macos"))]
171    fn refresh_io(&mut self, _refresh_kind: DiskRefreshKind) {}
172}

174impl crate::DisksInner {
175    pub(crate) fn new() -> Self {
176        Self {
177            disks: Vec::with_capacity(2),
178        }
179    }

181    pub(crate) fn refresh_specifics(
182        &mut self,
183        remove_not_listed_disks: bool,
184        refresh_kind: DiskRefreshKind,
185    ) {
186        unsafe {
187            // SAFETY: We don't keep any Objective-C objects around because we
188            // don't make any direct Objective-C calls in this code.
189            with_autorelease(|| {
190                get_list(&mut self.disks, refresh_kind);
191            })
192        }
193
194        if remove_not_listed_disks {
195            self.disks.retain_mut(|disk| {
196                if !disk.inner.updated {
197                    return false;
198                }
199                disk.inner.updated = false;
200                true
201            });
202        } else {
203            for c in self.disks.iter_mut() {
204                c.inner.updated = false;
205            }
206        }
207    }

209    pub(crate) fn list(&self) -> &[Disk] {
210        &self.disks
211    }
212
213    pub(crate) fn list_mut(&mut self) -> &mut [Disk] {
214        &mut self.disks
215    }

218unsafe fn get_list(container: &mut Vec<Disk>, refresh_kind: DiskRefreshKind) {
219    let raw_disks = {
220        let count = libc::getfsstat(ptr::null_mut(), 0, libc::MNT_NOWAIT);
221        if count < 1 {
222            return;
223        }
224        let bufsize = count * std::mem::size_of::<libc::statfs>() as libc::c_int;
225        let mut disks = Vec::with_capacity(count as _);
226        let count = libc::getfsstat(disks.as_mut_ptr(), bufsize, libc::MNT_NOWAIT);
227
228        if count < 1 {
229            return;
230        }
231
232        disks.set_len(count as usize);
233
234        disks
235    };
236
237    // Currently we query maximum 10 properties.
238    let mut properties = Vec::with_capacity(10);
239    // "mandatory" information
240    properties.push(kCFURLVolumeNameKey);
241    properties.push(kCFURLVolumeIsBrowsableKey);
242    properties.push(kCFURLVolumeIsLocalKey);
243    properties.push(kCFURLVolumeUUIDStringKey);
244
245    // is_removable
246    properties.push(kCFURLVolumeIsEjectableKey);
247    properties.push(kCFURLVolumeIsRemovableKey);
248    properties.push(kCFURLVolumeIsInternalKey);
249
250    if refresh_kind.storage() {
251        properties.push(kCFURLVolumeTotalCapacityKey);
252        properties.push(kCFURLVolumeAvailableCapacityForImportantUsageKey);
253        properties.push(kCFURLVolumeAvailableCapacityKey);
254    }
255
256    // Create a list of properties about the disk that we want to fetch.
257    let requested_properties = match build_requested_properties(&properties) {
258        Some(properties) => properties,
259        None => {
260            sysinfo_debug!("failed to create volume key list");
261            return;
262        }
263    };
264
265    for c_disk in raw_disks {
266        let volume_url = match CFURL::from_file_system_representation(
267            kCFAllocatorDefault,
268            c_disk.f_mntonname.as_ptr() as *const _,
269            c_disk.f_mntonname.len() as _,
270            false as _,
271        ) {
272            Some(url) => url,
273            None => {
274                sysinfo_debug!("getfsstat returned incompatible paths");
275                continue;
276            }
277        };
278
279        let prop_dict = match get_disk_properties(&volume_url, &requested_properties) {
280            Some(props) => props,
281            None => continue,
282        };
283
284        // Future note: There is a difference between `kCFURLVolumeIsBrowsableKey` and the
285        // `kCFURLEnumeratorSkipInvisibles` option of `CFURLEnumeratorOptions`. Specifically,
286        // the first one considers the writable `Data`(`/System/Volumes/Data`) partition to be
287        // browsable, while it is classified as "invisible" by CoreFoundation's volume enumerator.
288        let browsable = get_bool_value(&prop_dict, kCFURLVolumeIsBrowsableKey).unwrap_or_default();
289
290        // Do not return invisible "disks". Most of the time, these are APFS snapshots, hidden
291        // system volumes, etc. Browsable is defined to be visible in the system's UI like Finder,
292        // disk utility, system information, etc.
293        //
294        // To avoid seemingly duplicating many disks and creating an inaccurate view of the system's
295        // resources, these are skipped entirely.
296        if !browsable {
297            continue;
298        }
299
300        let local_only = get_bool_value(&prop_dict, kCFURLVolumeIsLocalKey).unwrap_or(true);
301
302        // Skip any drive that is not locally attached to the system.
303        //
304        // This includes items like SMB mounts, and matches the other platform's behavior.
305        if !local_only {
306            continue;
307        }
308
309        let mount_point = PathBuf::from(OsStr::from_bytes(
310            CStr::from_ptr(c_disk.f_mntonname.as_ptr()).to_bytes(),
311        ));
312
313        let uuid = OsStr::from_bytes(CStr::from_ptr(c_disk.f_mntonname.as_ptr()).to_bytes())
314            .to_os_string();
315
316        let disk = container.iter_mut().find(|d| d.inner.uuid == uuid);
317        if let Some(disk) = new_disk(
318            disk,
319            mount_point,
320            volume_url,
321            c_disk,
322            &prop_dict,
323            refresh_kind,
324            uuid,
325        ) {
326            container.push(disk);
327        }
328    }
329}

331unsafe fn build_requested_properties(
332    properties: &[Option<&CFString>],
333) -> Option<CFRetained<CFArray>> {
334    CFArray::new(
335        None,
336        properties.as_ptr() as *mut *const c_void,
337        properties.len() as _,
338        &kCFTypeArrayCallBacks,
339    )
340}

342fn get_disk_properties(
343    volume_url: &CFURL,
344    requested_properties: &CFArray,
345) -> Option<CFRetained<CFDictionary>> {
346    unsafe { volume_url.resource_properties_for_keys(Some(requested_properties), ptr::null_mut()) }
347}

349fn get_available_volume_space(disk_props: &CFDictionary) -> Option<u64> {
350    // We prefer `AvailableCapacityForImportantUsage` over `AvailableCapacity` because
351    // it takes more of the system's properties into account, like the trash, system-managed caches,
352    // etc. It generally also returns higher values too, because of the above, so it's a more
353    // accurate representation of what the system _could_ still use.
354    unsafe {
355        get_int_value(
356            disk_props,
357            kCFURLVolumeAvailableCapacityForImportantUsageKey,
358        )
359        .filter(|bytes| *bytes != 0)
360        .or_else(|| get_int_value(disk_props, kCFURLVolumeAvailableCapacityKey))
361    }
362}

364unsafe fn get_dict_value<T, F: FnOnce(*const c_void) -> Option<T>>(
365    dict: &CFDictionary,
366    key: Option<&CFString>,
367    callback: F,
368) -> Option<T> {
369    let mut value = std::ptr::null();
370    let key: *const CFString = key.map(|key| key as *const CFString).unwrap_or(ptr::null());
371    if dict.value_if_present(key.cast(), &mut value) {
372        callback(value)
373    } else {
374        None
375    }
376}
377
378pub(super) unsafe fn get_str_value(dict: &CFDictionary, key: Option<&CFString>) -> Option<String> {
379    get_dict_value(dict, key, |v| {
380        let v = unsafe { &*v.cast::<CFString>() };
381        Some(v.to_string())
382    })
383}
384
385unsafe fn get_bool_value(dict: &CFDictionary, key: Option<&CFString>) -> Option<bool> {
386    get_dict_value(dict, key, |v| {
387        let v = unsafe { &*v.cast::<CFBoolean>() };
388        Some(v.as_bool())
389    })
390}
391
392pub(super) unsafe fn get_int_value(dict: &CFDictionary, key: Option<&CFString>) -> Option<u64> {
393    get_dict_value(dict, key, |v| {
394        let v = unsafe { &*v.cast::<CFNumber>() };
395        Some(v.as_i64()? as u64)
396    })
397}

399unsafe fn new_disk(
400    disk: Option<&mut Disk>,
401    mount_point: PathBuf,
402    volume_url: CFRetained<CFURL>,
403    c_disk: libc::statfs,
404    disk_props: &CFDictionary,
405    refresh_kind: DiskRefreshKind,
406    uuid: OsString,
407) -> Option<Disk> {
408    let (total_space, available_space) = if refresh_kind.storage() {
409        (
410            get_int_value(disk_props, kCFURLVolumeTotalCapacityKey),
411            get_available_volume_space(disk_props),
412        )
413    } else {
414        (None, None)
415    };
416
417    // We update the existing disk here to prevent having another call to get `storage` info.
418    if let Some(disk) = disk {
419        let disk = &mut disk.inner;
420        if let Some(total_space) = total_space {
421            disk.total_space = total_space;
422        }
423        if let Some(available_space) = available_space {
424            disk.available_space = available_space;
425        }
426        disk.refresh_io(refresh_kind);
427        disk.refresh_kind(refresh_kind);
428        disk.updated = true;
429        return None;
430    }
431
432    // Note: Since we requested these properties from the system, we don't expect
433    // these property retrievals to fail.
434
435    let name = get_str_value(disk_props, kCFURLVolumeNameKey).map(OsString::from)?;
436
437    let file_system = {
438        let len = c_disk
439            .f_fstypename
440            .iter()
441            .position(|&b| b == 0)
442            .unwrap_or(c_disk.f_fstypename.len());
443        OsString::from_vec(
444            c_disk.f_fstypename[..len]
445                .iter()
446                .map(|c| *c as u8)
447                .collect(),
448        )
449    };
450
451    #[cfg(target_os = "macos")]
452    let bsd_name = get_bsd_name(&c_disk);
453
454    // IOKit is not available on any but the most recent (16+) iOS and iPadOS versions.
455    // Due to this, we can't query the medium type and disk i/o stats. All iOS devices use flash-based storage
456    // so we just assume the disk type is an SSD and set disk i/o stats to 0 until Rust has a way to conditionally link to
457    // IOKit in more recent deployment versions.
458
459    let ejectable = get_bool_value(disk_props, kCFURLVolumeIsEjectableKey).unwrap_or(false);
460
461    let removable = get_bool_value(disk_props, kCFURLVolumeIsRemovableKey).unwrap_or(false);
462
463    let is_removable = if ejectable || removable {
464        true
465    } else {
466        // If neither `ejectable` or `removable` return `true`, fallback to checking
467        // if the disk is attached to the internal system.
468        let internal = get_bool_value(disk_props, kCFURLVolumeIsInternalKey).unwrap_or_default();
469
470        !internal
471    };
472
473    let is_read_only = (c_disk.f_flags & libc::MNT_RDONLY as u32) != 0;
474
475    let mut disk = DiskInner {
476        type_: DiskKind::Unknown(-1),
477        name,
478        #[cfg(target_os = "macos")]
479        bsd_name,
480        file_system,
481        mount_point,
482        volume_url,
483        total_space: total_space.unwrap_or(0),
484        available_space: available_space.unwrap_or(0),
485        is_removable,
486        is_read_only,
487        read_bytes: 0,
488        written_bytes: 0,
489        old_read_bytes: 0,
490        old_written_bytes: 0,
491        updated: true,
492        uuid,
493    };
494
495    disk.refresh_kind(refresh_kind);
496    disk.refresh_io(refresh_kind);
497
498    Some(Disk { inner: disk })
499}

501/// Calls the provided closure in the context of a new autorelease pool that is drained
502/// before returning.
503///
504/// ## SAFETY:
505/// You must not return an Objective-C object that is autoreleased from this function since it
506/// will be freed before usable.
507unsafe fn with_autorelease<T, F: FnOnce() -> T>(call: F) -> T {
508    // NB: This struct exists to help prevent memory leaking if `call` were to panic.
509    // Otherwise, the call to `objc_autoreleasePoolPop` would never be made as the stack unwinds.
510    // `Drop` destructors for existing types on the stack are run during unwinding, so we can
511    // ensure the autorelease pool is drained by using a RAII pattern here.
512    struct DrainPool {
513        ctx: *mut c_void,
514    }
515
516    impl Drop for DrainPool {
517        fn drop(&mut self) {
518            // SAFETY: We have not manipulated `pool_ctx` since it was received from a corresponding
519            // pool push call.
520            unsafe { ffi::objc_autoreleasePoolPop(self.ctx) }
521        }
522    }
523
524    // SAFETY: Creating a new pool is safe in any context. They can be arbitrarily nested
525    // as long as pool objects are not used in deeper layers, but we only have one and don't
526    // allow it to leave this scope.
527    let _pool_ctx = DrainPool {
528        ctx: unsafe { ffi::objc_autoreleasePoolPush() },
529    };
530    call()
531    // Pool is drained here before returning
532}

534#[cfg(target_os = "macos")]
535fn get_bsd_name(disk: &libc::statfs) -> Option<Vec<u8>> {
536    // Removes `/dev/` from the value.
537    unsafe {
538        CStr::from_ptr(disk.f_mntfromname.as_ptr())
539            .to_bytes_with_nul()
540            .strip_prefix(b"/dev/")
541            .map(|slice| slice.to_vec())
542            .or_else(|| {
543                sysinfo_debug!("unknown disk mount path format");
544                None
545            })
546    }
