14pub(crate) struct CpusWrapper {
15    pub(crate) global_cpu: CpuUsage,
16    pub(crate) cpus: Vec<Cpu>,
17    pub(crate) got_cpu_frequency: bool,
18    /// This field is needed to prevent updating when not enough time passed since last update.
19    last_update: Option<Instant>,
20}
21
22impl CpusWrapper {
23    pub(crate) fn new() -> Self {
24        Self {
25            global_cpu: CpuUsage::new(),
26            cpus: Vec::new(),
27            got_cpu_frequency: false,
28            last_update: None,
29        }
30    }
31
32    pub(crate) fn refresh(&mut self, refresh_kind: CpuRefreshKind, port: mach_port_t) {
33        let need_cpu_usage_update = self
34            .last_update
35            .is_some_and(|last_update| last_update.elapsed() >= crate::MINIMUM_CPU_UPDATE_INTERVAL);
36
37        let cpus = &mut self.cpus;
38        if cpus.is_empty() {
39            init_cpus(port, cpus, &mut self.global_cpu, refresh_kind);
40            self.last_update = Some(Instant::now());
41            self.got_cpu_frequency = refresh_kind.frequency();
42            return;
43        }
44        if refresh_kind.frequency() && !self.got_cpu_frequency {
45            let frequency = unsafe { get_cpu_frequency(cpus.first().map_or("", |c| c.brand())) };
46            for proc_ in cpus.iter_mut() {
47                proc_.inner.set_frequency(frequency);
48            }
49            self.got_cpu_frequency = true;
50        }
51        if refresh_kind.cpu_usage() && need_cpu_usage_update {
52            self.last_update = Some(Instant::now());
53            update_cpu_usage(port, &mut self.global_cpu, |proc_data, cpu_info| {
54                let mut percentage = 0f32;
55                let mut offset = 0;
56                for proc_ in cpus.iter_mut() {
57                    let cpu_usage = compute_usage_of_cpu(proc_, cpu_info, offset);
58                    proc_.inner.update(cpu_usage, Arc::clone(&proc_data));
59                    percentage += proc_.inner.cpu_usage();
60
61                    offset += libc::CPU_STATE_MAX as isize;
62                }
63                (percentage, cpus.len())
64            });
65        }
66    }
67}

69pub(crate) struct UnsafePtr<T>(*mut T);
70
71unsafe impl<T> Send for UnsafePtr<T> {}
72unsafe impl<T> Sync for UnsafePtr<T> {}
73
74impl<T> Deref for UnsafePtr<T> {
75    type Target = *mut T;
76
77    fn deref(&self) -> &*mut T {
78        &self.0
79    }
80}

88    pub fn new(cpu_info: *mut i32, num_cpu_info: u32) -> CpuData {
89        CpuData {
90            cpu_info: UnsafePtr(cpu_info),
91            num_cpu_info,
92        }
93    }
94}
95
96impl Drop for CpuData {
97    fn drop(&mut self) {
98        if !self.cpu_info.0.is_null() {
99            let prev_cpu_info_size = std::mem::size_of::<i32>() as u32 * self.num_cpu_info;
100            unsafe {
101                libc::vm_deallocate(
102                    #[allow(deprecated)]
103                    mach_task_self(),
104                    self.cpu_info.0 as _,
105                    prev_cpu_info_size as _,
106                );
107            }
108            self.cpu_info.0 = std::ptr::null_mut();
109        }
110    }
111}

203pub(crate) unsafe fn get_cpu_frequency(#[allow(unused_variables)] brand: &str) -> u64 {
204    let mut speed: u64 = 0;
205    let mut len = std::mem::size_of::<u64>();
206    if libc::sysctlbyname(
207        b"hw.cpufrequency\0".as_ptr() as *const _,
208        &mut speed as *mut _ as _,
209        &mut len,
210        std::ptr::null_mut(),
211        0,
212    ) == 0
213    {
214        return speed / 1_000_000;
215    }
216
217    #[cfg(any(target_os = "ios", feature = "apple-sandbox"))]
218    {
219        0
220    }
221    #[cfg(not(any(target_os = "ios", feature = "apple-sandbox")))]
222    {
223        crate::sys::inner::cpu::get_cpu_frequency(brand)
224    }
225}

243#[inline]
244fn get_in_use(cpu_info: *mut i32, offset: isize) -> i64 {
245    unsafe {
246        let user = *cpu_info.offset(offset + libc::CPU_STATE_USER as isize) as i64;
247        let system = *cpu_info.offset(offset + libc::CPU_STATE_SYSTEM as isize) as i64;
248        let nice = *cpu_info.offset(offset + libc::CPU_STATE_NICE as isize) as i64;
249
250        user.saturating_add(system).saturating_add(nice)
251    }
252}
253
254#[inline]
255fn get_idle(cpu_info: *mut i32, offset: isize) -> i32 {
256    unsafe { *cpu_info.offset(offset + libc::CPU_STATE_IDLE as isize) }
257}

259pub(crate) fn compute_usage_of_cpu(proc_: &Cpu, cpu_info: *mut i32, offset: isize) -> f32 {
260    let old_cpu_info = proc_.inner.data().cpu_info.0;
261    let in_use;
262    let idle;
263
264    // In case we are initializing cpus, there is no "old value" yet.
265    if std::ptr::eq(old_cpu_info, cpu_info) {
266        in_use = get_in_use(cpu_info, offset);
267        idle = get_idle(cpu_info, offset);
268    } else {
269        let new_in_use = get_in_use(cpu_info, offset);
270        let old_in_use = get_in_use(old_cpu_info, offset);
271
272        let new_idle = get_idle(cpu_info, offset);
273        let old_idle = get_idle(old_cpu_info, offset);
274
275        in_use = new_in_use.saturating_sub(old_in_use);
276        idle = new_idle.saturating_sub(old_idle) as _;
277    }
278    let total = in_use.saturating_add(idle as _);
279    let usage = (in_use as f32 / total as f32) * 100.;
280    if usage.is_nan() {
281        // If divided by zero, avoid returning a NaN
282        0.
283    } else {
284        usage
285    }
286}

288pub(crate) fn update_cpu_usage<F: FnOnce(Arc<CpuData>, *mut i32) -> (f32, usize)>(
289    port: libc::mach_port_t,
290    global_cpu: &mut CpuUsage,
291    f: F,
292) {
293    let mut num_cpu_u = 0u32;
294    let mut cpu_info: *mut i32 = std::ptr::null_mut();
295    let mut num_cpu_info = 0u32;
296
297    let mut total_cpu_usage = 0f32;
298
299    unsafe {
300        if host_processor_info(
301            port,
302            libc::PROCESSOR_CPU_LOAD_INFO,
303            &mut num_cpu_u as *mut u32,
304            &mut cpu_info as *mut *mut i32,
305            &mut num_cpu_info as *mut u32,
306        ) == libc::KERN_SUCCESS
307        {
308            let (total_percentage, len) =
309                f(Arc::new(CpuData::new(cpu_info, num_cpu_info)), cpu_info);
310            total_cpu_usage = total_percentage / len as f32;
311        }
312        global_cpu.set_cpu_usage(total_cpu_usage);
313    }
314}

316pub(crate) fn init_cpus(
317    port: libc::mach_port_t,
318    cpus: &mut Vec<Cpu>,
319    global_cpu: &mut CpuUsage,
320    refresh_kind: CpuRefreshKind,
321) {
322    let mut num_cpu = 0;
323    let mut mib = [libc::CTL_HW as _, libc::HW_NCPU as _];
324
325    let (vendor_id, brand) = get_vendor_id_and_brand();
326    let frequency = if refresh_kind.frequency() {
327        unsafe { get_cpu_frequency(&brand) }
328    } else {
329        global_cpu.frequency
330    };
331
332    unsafe {
333        if !get_sys_value(
334            mem::size_of::<u32>(),
335            &mut num_cpu as *mut _ as *mut _,
336            &mut mib,
337        ) {
338            num_cpu = 1;
339        }
340    }
341    update_cpu_usage(port, global_cpu, |proc_data, cpu_info| {
342        let mut percentage = 0f32;
343        let mut offset = 0;
344        for i in 0..num_cpu {
345            let mut cpu = Cpu {
346                inner: CpuInner::new(
347                    format!("{}", i + 1),
348                    Arc::clone(&proc_data),
349                    frequency,
350                    vendor_id.clone(),
351                    brand.clone(),
352                ),
353            };
354            if refresh_kind.cpu_usage() {
355                let cpu_usage = compute_usage_of_cpu(&cpu, cpu_info, offset);
356                cpu.inner.set_cpu_usage(cpu_usage);
357                percentage += cpu.cpu_usage();
358            }
359            cpus.push(cpu);
360
361            offset += libc::CPU_STATE_MAX as isize;
362        }
363        (percentage, cpus.len())
364    });
365}

367pub(crate) fn get_sysctl_str(s: &[u8]) -> String {
368    let mut len = 0;
369
370    unsafe {
371        libc::sysctlbyname(
372            s.as_ptr() as *const c_char,
373            std::ptr::null_mut(),
374            &mut len,
375            std::ptr::null_mut(),
376            0,
377        );
378        if len < 1 {
379            return String::new();
380        }
381
382        let mut buf = Vec::with_capacity(len);
383        libc::sysctlbyname(
384            s.as_ptr() as *const c_char,
385            buf.as_mut_ptr() as _,
386            &mut len,
387            std::ptr::null_mut(),
388            0,
389        );
390        if len > 0 {
391            buf.set_len(len);
392            while buf.last() == Some(&b'\0') {
393                buf.pop();
394            }
395            String::from_utf8(buf).unwrap_or_else(|_| String::new())
396        } else {
397            String::new()
398        }
399    }
400}

402pub(crate) fn get_vendor_id_and_brand() -> (String, String) {

404    // is empty.

406    if vendor.is_empty() {
407        vendor = "Apple".to_string();
408    }
409
410    let brand = get_sysctl_str(b"machdep.cpu.brand_string\0");

463#[cfg(test)]
464mod test {
465    use crate::*;
466    use std::process::Command;
467
468    #[test]
469    fn check_vendor_and_brand() {
470        let child = Command::new("sysctl")
471            .arg("-a")
472            .output()
473            .expect("Failed to start command...");
474
475        assert!(child.status.success());
476        let stdout = String::from_utf8(child.stdout).expect("Not valid UTF8");
477
478        let sys = System::new_with_specifics(
479            crate::RefreshKind::nothing().with_cpu(CpuRefreshKind::nothing().with_cpu_usage()),
480        );
481        let cpus = sys.cpus();
482        assert!(!cpus.is_empty(), "no CPU found");
483        if let Some(line) = stdout.lines().find(|l| l.contains("machdep.cpu.vendor")) {
484            let sysctl_value = line.split(':').nth(1).unwrap();
485            assert_eq!(cpus[0].vendor_id(), sysctl_value.trim());
486        }
487        if let Some(line) = stdout
488            .lines()
489            .find(|l| l.contains("machdep.cpu.brand_string"))
490        {
491            let sysctl_value = line.split(':').nth(1).unwrap();
492            assert_eq!(cpus[0].brand(), sysctl_value.trim());
493        }
494    }
