99fn boot_time() -> u64 {
100    let mut boot_time = timeval {
101        tv_sec: 0,
102        tv_usec: 0,
103    };
104    let mut len = std::mem::size_of::<timeval>();
105    let mut mib: [c_int; 2] = [libc::CTL_KERN, libc::KERN_BOOTTIME];
106
107    unsafe {
108        if sysctl(
109            mib.as_mut_ptr(),
110            mib.len() as _,
111            &mut boot_time as *mut timeval as *mut _,
112            &mut len,
113            std::ptr::null_mut(),
114            0,
115        ) < 0
116        {
117            0
118        } else {
119            boot_time.tv_sec as _
120        }
121    }
122}

155    pub(crate) fn refresh_memory_specifics(&mut self, refresh_kind: MemoryRefreshKind) {
156        let mut mib = [libc::CTL_VM as _, libc::VM_SWAPUSAGE as _];
157
158        unsafe {
159            if refresh_kind.swap() {
160                // get system values
161                // get swap info
162                let mut xs: libc::xsw_usage = mem::zeroed::<libc::xsw_usage>();
163                if get_sys_value(
164                    mem::size_of::<libc::xsw_usage>(),
165                    &mut xs as *mut _ as *mut c_void,
166                    &mut mib,
167                ) {
168                    self.swap_total = xs.xsu_total;
169                    self.swap_free = xs.xsu_avail;
170                }
171            }
172            if refresh_kind.ram() {
173                mib[0] = libc::CTL_HW as _;
174                mib[1] = libc::HW_MEMSIZE as _;
175                // get ram info
176                if self.mem_total < 1 {
177                    get_sys_value(
178                        mem::size_of::<u64>(),
179                        &mut self.mem_total as *mut u64 as *mut c_void,
180                        &mut mib,
181                    );
182                }
183                let mut count: u32 = libc::HOST_VM_INFO64_COUNT as _;
184                let mut stat = mem::zeroed::<vm_statistics64>();
185                if host_statistics64(
186                    self.port,
187                    libc::HOST_VM_INFO64,
188                    &mut stat as *mut vm_statistics64 as *mut _,
189                    &mut count,
190                ) == libc::KERN_SUCCESS
191                {
192                    // From the apple documentation:
193                    //
194                    // /*
195                    //  * NB: speculative pages are already accounted for in "free_count",
196                    //  * so "speculative_count" is the number of "free" pages that are
197                    //  * used to hold data that was read speculatively from disk but
198                    //  * haven't actually been used by anyone so far.
199                    //  */
200                    self.mem_available = u64::from(stat.free_count)
201                        .saturating_add(u64::from(stat.inactive_count))
202                        .saturating_add(u64::from(stat.purgeable_count))
203                        .saturating_sub(u64::from(stat.compressor_page_count))
204                        .saturating_mul(self.page_size_b);
205                    self.mem_used = u64::from(stat.active_count)
206                        .saturating_add(u64::from(stat.wire_count))
207                        .saturating_add(u64::from(stat.compressor_page_count))
208                        .saturating_add(u64::from(stat.speculative_count))
209                        .saturating_mul(self.page_size_b);
210                    self.mem_free = u64::from(stat.free_count)
211                        .saturating_sub(u64::from(stat.speculative_count))
212                        .saturating_mul(self.page_size_b);
213                }
214            }
215        }
216    }

240    #[cfg(all(target_os = "macos", not(feature = "apple-sandbox")))]
241    pub(crate) fn refresh_processes_specifics(
242        &mut self,
243        processes_to_update: ProcessesToUpdate<'_>,
244        refresh_kind: ProcessRefreshKind,
245    ) -> usize {
246        use crate::utils::into_iter;
247        use std::sync::atomic::{AtomicUsize, Ordering};
248
249        unsafe {
250            let count = libc::proc_listallpids(::std::ptr::null_mut(), 0);
251            if count < 1 {
252                return 0;
253            }
254        }
255        if let Some(pids) = get_proc_list() {
256            #[inline(always)]
257            fn real_filter(e: Pid, filter: &[Pid]) -> bool {
258                filter.contains(&e)
259            }
260
261            #[inline(always)]
262            fn empty_filter(_e: Pid, _filter: &[Pid]) -> bool {
263                true
264            }
265
266            #[allow(clippy::type_complexity)]
267            let (filter, filter_callback): (
268                &[Pid],
269                &(dyn Fn(Pid, &[Pid]) -> bool + Sync + Send),
270            ) = match processes_to_update {
271                ProcessesToUpdate::All => (&[], &empty_filter),
272                ProcessesToUpdate::Some(pids_to_refresh) => {
273                    if pids_to_refresh.is_empty() {
274                        return 0;
275                    }
276                    (pids_to_refresh, &real_filter)
277                }
278            };
279
280            let nb_updated = AtomicUsize::new(0);
281            let now = get_now();
282            let port = self.port;
283            let time_interval = self.clock_info.as_mut().map(|c| c.get_time_interval(port));
284            let timebase_to_ms = self
285                .clock_info
286                .as_ref()
287                .map(|c| c.timebase_to_ms)
288                .unwrap_or_default();
289            let entries: Vec<Process> = {
290                let wrap = &Wrap(UnsafeCell::new(&mut self.process_list));
291
292                #[cfg(feature = "multithread")]
293                use rayon::iter::ParallelIterator;
294
295                into_iter(pids)
296                    .flat_map(|pid| {
297                        if !filter_callback(pid, filter) {
298                            return None;
299                        }
300                        nb_updated.fetch_add(1, Ordering::Relaxed);
301                        update_process(
302                            wrap,
303                            pid,
304                            time_interval,
305                            now,
306                            refresh_kind,
307                            false,
308                            timebase_to_ms,
309                        )
310                        .unwrap_or_default()
311                    })
312                    .collect()
313            };
314            entries.into_iter().for_each(|entry| {
315                self.process_list.insert(entry.pid(), entry);
316            });
317            nb_updated.into_inner()
318        } else {
319            0
320        }
321    }

405    pub(crate) fn long_os_version() -> Option<String> {
406        #[cfg(target_os = "macos")]
407        {
408            let Some(os_version) = Self::os_version() else {
409                return Some("macOS".to_owned());
410            };
411            // https://en.wikipedia.org/wiki/MacOS_version_history
412            for (version_prefix, macos_spelling, friendly_name) in [
413                ("15", "macOS", "Sequoia"),
414                ("14", "macOS", "Sonoma"),
415                ("13", "macOS", "Ventura"),
416                ("12", "macOS", "Monterey"),
417                ("11", "macOS", "Big Sur"),
418                // Big Sur identifies itself as 10.16 in some situations.
419                // https://en.wikipedia.org/wiki/MacOS_Big_Sur#Development_history
420                ("10.16", "macOS", "Big Sur"),
421                ("10.15", "macOS", "Catalina"),
422                ("10.14", "macOS", "Mojave"),
423                ("10.13", "macOS", "High Sierra"),
424                ("10.12", "macOS", "Sierra"),
425                ("10.11", "OS X", "El Capitan"),
426                ("10.10", "OS X", "Yosemite"),
427                ("10.9", "OS X", "Mavericks"),
428                ("10.8", "OS X", "Mountain Lion"),
429                ("10.7", "Mac OS X", "Lion"),
430                ("10.6", "Mac OS X", "Snow Leopard"),
431                ("10.5", "Mac OS X", "Leopard"),
432                ("10.4", "Mac OS X", "Tiger"),
433                ("10.3", "Mac OS X", "Panther"),
434                ("10.2", "Mac OS X", "Jaguar"),
435                ("10.1", "Mac OS X", "Puma"),
436                ("10.0", "Mac OS X", "Cheetah"),
437            ] {
438                if os_version.starts_with(version_prefix) {
439                    return Some(format!("{macos_spelling} {os_version} {friendly_name}"));
440                }
441            }
442            Some(format!("macOS {os_version}"))
443        }
444
445        #[cfg(target_os = "ios")]
446        {
447            let mut long_name = "iOS".to_owned();
448            if let Some(os_version) = Self::os_version() {
449                long_name.push(' ');
450                long_name.push_str(&os_version);
451            }
452            Some(long_name)
453        }
454    }

464    pub(crate) fn os_version() -> Option<String> {
465        unsafe {
466            // get the size for the buffer first
467            let mut size = 0;
468            if get_sys_value_by_name(b"kern.osproductversion\0", &mut size, std::ptr::null_mut())
469                && size > 0
470            {
471                // now create a buffer with the size and get the real value
472                let mut buf = vec![0_u8; size as _];
473
474                if get_sys_value_by_name(
475                    b"kern.osproductversion\0",
476                    &mut size,
477                    buf.as_mut_ptr() as *mut c_void,
478                ) {
479                    if let Some(pos) = buf.iter().position(|x| *x == 0) {
480                        // Shrink buffer to terminate the null bytes
481                        buf.resize(pos, 0);
482                    }
483
484                    String::from_utf8(buf).ok()
485                } else {
486                    // getting the system value failed
487                    None
488                }
489            } else {
490                // getting the system value failed, or did not return a buffer size
491                None
492            }
493        }
494    }

508    pub(crate) fn cpu_arch() -> Option<String> {
509        let mut arch_str: [u8; 32] = [0; 32];
510        let mut mib = [libc::CTL_HW as _, libc::HW_MACHINE as _];
511
512        unsafe {
513            if get_sys_value(
514                mem::size_of::<[u8; 32]>(),
515                arch_str.as_mut_ptr() as *mut _,
516                &mut mib,
517            ) {
518                CStr::from_bytes_until_nul(&arch_str)
519                    .ok()
520                    .and_then(|res| match res.to_str() {
521                        Ok(arch) => Some(arch.to_string()),
522                        Err(_) => None,
523                    })
524            } else {
525                None
526            }
527        }
528    }

547fn get_system_info(value: c_int, default: Option<&str>) -> Option<String> {
548    let mut mib: [c_int; 2] = [libc::CTL_KERN, value];
549    let mut size = 0;
550
551    unsafe {
552        // Call first to get size
553        sysctl(
554            mib.as_mut_ptr(),
555            mib.len() as _,
556            std::ptr::null_mut(),
557            &mut size,
558            std::ptr::null_mut(),
559            0,
560        );
561
562        // exit early if we did not update the size
563        if size == 0 {
564            default.map(|s| s.to_owned())
565        } else {
566            // set the buffer to the correct size
567            let mut buf = vec![0_u8; size as _];
568
569            if sysctl(
570                mib.as_mut_ptr(),
571                mib.len() as _,
572                buf.as_mut_ptr() as _,
573                &mut size,
574                std::ptr::null_mut(),
575                0,
576            ) == -1
577            {
578                // If command fails return default
579                default.map(|s| s.to_owned())
580            } else {
581                if let Some(pos) = buf.iter().position(|x| *x == 0) {
582                    // Shrink buffer to terminate the null bytes
583                    buf.resize(pos, 0);
584                }
585
586                String::from_utf8(buf).ok()
587            }
588        }
589    }
