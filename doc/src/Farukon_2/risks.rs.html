5/// Checks if sufficient capital exists to execute a signal.
6/// Used during SIGNAL → ORDER conversion.
7/// # Arguments
8/// * `quantity` - The quantity to trade.
9/// * `latest_equity_point` - The latest equity point.
10/// * `signal_event` - The signal event.
11/// * `instrument_info` - The instrument metadata.
12/// # Returns
13/// * `anyhow::Result<bool>` indicating whether the signal can be executed.
14pub fn margin_call_control_for_signal(
15    quantity: f64,
16    latest_holdings: &farukon_core::portfolio::HoldingSnapshot,
17    signal_event: &farukon_core::event::SignalEvent,
18    instrument_info: &farukon_core::instruments_info::InstrumentInfo,
19) -> anyhow::Result<bool> {
20    // Checks if sufficient capital exists to open a new position.
21    // Used during SIGNAL → ORDER conversion.
22
23    let signal_name = &signal_event.signal_name;
24    
25    if signal_name != "EXIT" {
26        let instrument_type = &instrument_info.instrument_type;
27        
28        if instrument_type == "futures" {
29            let margin = instrument_info.margin;
30            let initial_margin = quantity * margin;
31            let capital = latest_holdings.capital;
32
33            if capital > initial_margin {
34                return anyhow::Ok(true);
35            } else {
36                return anyhow::Ok(false);
37            }
38        }
39    } else {
40        return anyhow::Ok(true);    // EXIT always allowed
41    }
42
43    anyhow::Ok(true)
44}

46/// Checks if current portfolio has sufficient equity to maintain open positions.
47/// Triggers margin call if capital < min_margin * total_position_value.
48/// # Arguments
49/// * `latest_equity_point` - The latest equity point.
50/// * `current_positions` - The current positions.
51/// * `strategy_settings` - The strategy settings.
52/// * `strategy_instruments_info` - The instrument metadata.
53/// # Returns
54/// * `anyhow::Result<bool>` indicating whether a margin call has occurred.
55pub fn margin_call_control_for_market(
56    latest_holdings: &farukon_core::portfolio::HoldingSnapshot,
57    current_positions: &std::collections::HashMap<String, farukon_core::portfolio::PositionState>,
58    strategy_settings: &farukon_core::settings::StrategySettings,
59    strategy_instruments_info: &std::collections::HashMap<String, farukon_core::instruments_info::InstrumentInfo>,
60) -> anyhow::Result<bool> {
61    // Checks if current portfolio has sufficient equity to maintain open positions.
62    // Triggers margin call if capital < min_margin * total_position_value.
63
64    let cash = latest_holdings.cash;
65    if cash < 0.0 {
66        let mut capital = 0.0;
67        
68        for symbol in &strategy_settings.symbols {
69            let instrument_info = strategy_instruments_info.get(symbol).unwrap();
70            if instrument_info.instrument_type == "futures" {
71                let current_position_for_symbol = current_positions.get(symbol).unwrap().position;
72                if current_position_for_symbol != 0.0 {
73                    let entry_capital_for_symbol = current_positions.get(symbol).unwrap().entry_capital;
74                    capital += entry_capital_for_symbol;
75                }
76            }
77        }
78        let min_margin_param = strategy_settings.margin_params.min_margin;
79        let min_margin_for_strategy = capital * min_margin_param;
80        let strategy_current_capital = latest_holdings.capital;
81
82        if strategy_current_capital < min_margin_for_strategy {
83            return anyhow::Ok(false);
84        } else {
85            return anyhow::Ok(true);
86        }
87    }
88
89    anyhow::Ok(true)
90}
