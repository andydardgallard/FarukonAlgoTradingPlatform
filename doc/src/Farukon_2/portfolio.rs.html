77    /// Constructs the initial position state for each symbol in the strategy.
78    fn construct_current_positions(
79        strategy_settings: &farukon_core::settings::StrategySettings,
80    ) -> std::collections::HashMap<String, farukon_core::portfolio::PositionState> {
81        // Initializes empty position state for each symbol in strategy.
82
83        let mut positions: std::collections::HashMap<String, farukon_core::portfolio::PositionState> = std::collections::HashMap::new();
84        for symbol in &strategy_settings.symbols {
85            let current_positions_for_symbol: farukon_core::portfolio::PositionState = farukon_core::portfolio::PositionState::new();

105    /// Generates an order event from a signal event.
106    /// Applies position sizing and margin control.
107    /// # Arguments
108    /// * `signal_event` - The signal event received from the strategy.
109    /// * `data_handler` - The data handler for accessing market data.
110    /// # Returns
111    /// * An optional OrderEvent if the signal should be executed.
112    fn generate_order(
113        &self,
114        signal_event: &farukon_core::event::SignalEvent,
115        data_handler: &Box<dyn farukon_core::data_handler::DataHandler>,
116    ) -> Option<farukon_core::event::OrderEvent> {
117            // Converts a SIGNAL event into an ORDER event.
118            // Uses position sizer to determine quantity.
119            // Applies margin control to prevent over-leverage.
120
121            let signal_name = &signal_event.signal_name;
122            let symbol = &signal_event.symbol;
123            let cur_quantity = self.get_current_positions().get(&symbol.clone()).unwrap().position;
124            let cash = self.get_latest_holdings().unwrap().cash;
125            let current_datetime = data_handler.get_latest_bar_datetime(symbol).unwrap();
126            let order_type = &signal_event.order_type;
127            let limit_price = signal_event.limit_price;
128
129            let mut quantity = signal_event.quantity.unwrap().abs();
130            let mut direction = None;
131
132            if signal_name != "EXIT" && cur_quantity == 0.0 && cash > 0.0 {
133                quantity = farukon_core::utils::calculate_max_available_quantity(
134                    cash,
135                    quantity,
136                    self.strategy_instruments_info.get(symbol).unwrap());
137                
138                if signal_name == "LONG"  {
139                direction = Some("BUY".to_string());
140                } else if signal_name == "SHORT" {
141                direction = Some("SELL".to_string());
142            }
143            } else {
144                if cur_quantity > 0.0 {
145                direction = Some("SELL".to_string());
146                } else if cur_quantity < 0.0 {
147                    direction = Some("BUY".to_string());
148            }
149            }
150
151            if risks::margin_call_control_for_signal(
152                quantity,
153                self.get_latest_holdings().unwrap(),
154                signal_event, self.strategy_instruments_info.get(symbol).unwrap(),
155            ).ok()? {
156                let order = Some(farukon_core::event::OrderEvent::new(
157                    current_datetime,
158                    symbol.to_string(),
159                    order_type.to_string(),
160                    quantity,
161                    direction,
162                    signal_name.to_string(),
163                    limit_price,
164                ));
165
166                return order;
167            }
168
169            None
170        }
171
172}

175    /// Updates position state based on a fill event.
176    /// # Arguments
177    /// * `fill_event` - The fill event received from the execution engine.
178    /// * `data_handler` - The data handler for accessing market data.
179    fn update_positions_from_fill(
180        &mut self,
181        fill_event: &farukon_core::event::FillEvent,
182        data_handler: &Box<dyn farukon_core::data_handler::DataHandler>
183    ) {
184        // Updates position state on FILL.
185        // Adjusts position size, records entry/exit prices.
186
187        let timeindex = data_handler.get_latest_bar_datetime(&fill_event.symbol);
188        if self.mode == "Debug".to_string() {
189            println!("for Fill event, start Current_positions, {:?}, {:?}", timeindex, self.current_positions);
190        }
191
192        let fill_dir = match fill_event.direction.as_deref() {
193            Some("BUY") => 1.0,
194            Some("SELL") => -1.0,
195            _ => {
196                eprintln!("Unknown fill direction: {}", fill_event.direction.clone().unwrap());
197                0.0
198            },
199        };
200        let symbol = &fill_event.symbol;
201        let quantity = fill_event.quantity;
202        let signal_name = fill_event.signal_name.as_str();
203        let current_cash  = self.get_latest_holdings().unwrap().cash;
204
205        if let Some(position_state) = self.current_positions.get_mut(symbol) {
206            position_state.position += fill_dir * quantity;
207
208            match signal_name {
209                "EXIT" => {
210                    position_state.entry_price = None;
211                    position_state.entry_capital = 0.0;
212                },
213                _ => {
214                    position_state.deal_number += 1;
215                    position_state.entry_price = fill_event.execution_price;
216                    position_state.entry_capital = current_cash;
217                },
218            }
219        }
220
221        if self.mode == "Debug".to_string() {
222            println!("for Fill event, finish Current_positions, {:?}, {:?}", timeindex, self.current_positions);
223        }
224    }

226    /// Updates holding state based on a fill event.
227    /// # Arguments
228    /// * `fill_event` - The fill event received from the execution engine.
229    /// * `data_handler` - The data handler for accessing market data.
230    fn update_holdings_from_fill(
231        &mut self,
232        fill_event: &farukon_core::event::FillEvent,
233        data_handler: &Box<dyn farukon_core::data_handler::DataHandler>,
234    ) {
235        // Updates PnL and blocked margin on FILL.
236        // Uses step_price and step to convert price to currency value.
237
238        let timeindex = data_handler.get_latest_bar_datetime(&fill_event.symbol);
239        if self.mode == "Debug".to_string() {
240            println!("for Fill event, start Current_holdings, {:?}, {:?}", timeindex, self.current_holdings);
241        }
242        
243        let quantity = fill_event.quantity;
244        let symbol = &fill_event.symbol;
245        let commission = fill_event.commission.unwrap();
246        let signal_name = fill_event.signal_name.as_str();
247        let direction = fill_event.direction.clone().unwrap();
248        let execution_price = fill_event.execution_price.unwrap_or(0.0);
249        let close = match data_handler.get_latest_bar_value(symbol, "close") {
250            Some(value) if value.is_nan() => 0.0,
251            Some(value) => value,
252            None => 0.0,
253        };
254
255        let last_close = match data_handler.get_latest_bars_values(symbol, "close", 2).unwrap().first() {
256            Some(value) if value.is_nan() => 0.0,
257            Some(value) => *value,
258            None => 0.0,
259        };
260
261        let strategy_instrument_info_for_symbol = self.strategy_instruments_info.get(symbol).unwrap();
262        let step_price = strategy_instrument_info_for_symbol.step_price;
263        let step = strategy_instrument_info_for_symbol.step;
264        let cost_of_step_price = ((step_price / step) * 100_000.0).round() / 100_000.0;
265      
266        self.current_holdings.get_mut(symbol).unwrap().signal_name = Some(fill_event.signal_name.clone());
267        
268        match signal_name {
269            "EXIT" => {
270                match  direction.as_str() {
271                    "BUY" => {
272                        self.current_holdings.get_mut(symbol).unwrap().pnl = ((((last_close - execution_price) * cost_of_step_price) * quantity * 100.0).round() / 100.0) - commission;
273                    },
274                    "SELL" => {
275                        self.current_holdings.get_mut(symbol).unwrap().pnl = ((((execution_price - last_close) * cost_of_step_price) * quantity * 100.0).round() / 100.0) - commission;
276                    },
277                    _ => {
278                        eprintln!("Unknown fill direction: {:?}", fill_event.direction);
279                        return;
280                    }
281                }
282
283                match self.strategy_instruments_info.get(symbol).unwrap().instrument_type.as_str() {
284                    "futures" => {
285                        self.current_holdings.get_mut(symbol).unwrap().blocked -= strategy_instrument_info_for_symbol.margin * quantity;
286                    }
287                    _ => {
288                        eprintln!("Unknown type of instrument!");
289                    }
290                }
291            },
292            _ => {
293                match direction.as_str() {
294                    "BUY" => {
295                        self.current_holdings.get_mut(symbol).unwrap().pnl = ((((close - execution_price) * cost_of_step_price) * quantity * 100.0).round() / 100.0) - commission;
296                    },
297                    "SELL" => {
298                        self.current_holdings.get_mut(symbol).unwrap().pnl = ((((execution_price - close) * cost_of_step_price) * quantity * 100.0).round() / 100.0) - commission;
299                    },
300                    _ => {
301                        eprintln!("Unknown fill direction: {:?}", fill_event.direction);
302                        return;
303                    }
304                }
305
306                match self.strategy_instruments_info.get(symbol).unwrap().instrument_type.as_str() {
307                    "futures" => {
308                        self.current_holdings.get_mut(symbol).unwrap().blocked += strategy_instrument_info_for_symbol.margin * quantity;
309                    }
310                    _ => {
311                        eprintln!("Unknown type of instrument!");
312                    }
313                }
314            }
315        }
316
317        if self.mode == "Debug".to_string() {
318            println!("for Fill event, finish Current_holdings, {:?}, {:?}", timeindex, self.current_holdings);
319        }
320    }

335    /// Updates the portfolio state based on a fill event.
336    /// # Arguments
337    /// * `fill_event` - The fill event received from the execution engine.
338    /// * `data_handler` - The data handler for accessing market data.
339    fn update_timeindex(
340        &mut self,
341        data_handler: &Box<dyn farukon_core::data_handler::DataHandler>,
342    ) {
343        // Called on every MARKET event to update equity curve, snapshots, and metrics.
344        // Also triggers margin call monitoring.
345
346        let current_bar_datetime = data_handler.get_latest_bar_datetime(
347            &self.strategy_settings.symbols[0]
348            ).unwrap();
349
350        if self.mode == "Debug".to_string() {
351            println!(
352                "Start event, Update timeindex, {}, {:?}, {:?}",
353                current_bar_datetime,
354                self.current_positions,
355                self.current_holdings,
356            );
357        }
358
359        // Update all_positions snapshot
360        {
361            if self.all_positions.len() < 2 {
362                self.all_positions.push(farukon_core::portfolio::PositionSnapshot::new(
363                    current_bar_datetime,
364                    self.current_positions.clone(),
365                ));
366            } else {
367                if let Some(last) = self.all_positions.last_mut() {
368                    *last = farukon_core::portfolio::PositionSnapshot::new(
369                        current_bar_datetime,
370                        self.current_positions.clone(),
371                    );
372                }
373            }
374        }
375
376        // Update unrealized PnL for open positions
377        {
378            for symbol in &self.strategy_settings.symbols {
379                let close = match data_handler.get_latest_bar_value(symbol, "close") {
380                    Some(value) if value.is_nan() => 0.0,
381                    Some(value) => value,
382                    None => 0.0,
383                };
384
385                let last_close = match data_handler.get_latest_bars_values(symbol, "close", 2).unwrap().first() {
386                    Some(value) if value.is_nan() => 0.0,
387                    Some(value) => *value,
388                    None => 0.0,
389                };
390
391                let strategy_instrument_info_for_symbol = self.strategy_instruments_info.get(symbol).unwrap();
392                let step_price = strategy_instrument_info_for_symbol.step_price;
393                let step = strategy_instrument_info_for_symbol.step;
394                let cost_of_step_price = ((step_price / step) * 100_000.0).round() / 100_000.0;
395                
396                if let Some(holdings_state) = self.current_holdings.get_mut(symbol) {
397                    if let Some(position_state) = self.current_positions.get(symbol) {
398                        if holdings_state.signal_name != None {
399                            holdings_state.signal_name = None;
400                        } else {
401                            holdings_state.pnl = (((close - last_close) * cost_of_step_price) * position_state.position * 100.0).round() / 100.0;
402                        }
403                    }
404                }
405            }
406        }
407
408        // Update all_holdings snapshot
409        {
410            if self.all_holdings.len() == 0 {
411                let blocked = 0.0;
412                self.all_holdings.push(farukon_core::portfolio::HoldingSnapshot::new(
413                    current_bar_datetime,
414                    self.initial_capital_for_strategy,
415                    self.initial_capital_for_strategy,
416                    blocked, 
417                    self.current_holdings.clone(),
418                ));
419            } else if self.all_holdings.len() == 1 {
420                if let Some(first) = self.all_holdings.first_mut() {
421                    let blocked: f64 = self.current_holdings.iter()
422                        .map(|a| a.1.blocked)
423                        .sum();
424    
425                    let pnl: f64 = self.current_holdings.iter()
426                        .map(|a| a.1.pnl)
427                        .sum();
428    
429                    let capital = first.capital + pnl;
430                    let cash = capital - blocked;
431
432                    self.all_holdings.push(farukon_core::portfolio::HoldingSnapshot::new(
433                        current_bar_datetime,
434                        capital,
435                        cash,
436                        blocked,
437                        self.current_holdings.clone(),
438                    ));
439                }
440            } else {
441                if let Some(last) = self.all_holdings.last_mut() {
442                    let blocked: f64 = self.current_holdings.iter()
443                        .map(|a| a.1.blocked)
444                        .sum();
445    
446                    let pnl: f64 = self.current_holdings.iter()
447                        .map(|a| a.1.pnl)
448                        .sum();
449    
450                    let capital = last.capital + pnl;
451                    let cash = capital - blocked;
452    
453                    *last = farukon_core::portfolio::HoldingSnapshot::new(
454                        current_bar_datetime,
455                        capital,
456                        cash,
457                        blocked,
458                        self.current_holdings.clone(),
459                    );
460                }
461            }
462        }
463
464        // Udate equity curve data
465        {
466            if let Some(latest_holdings) = self.get_latest_holdings() {
467                self.equity_series.push((current_bar_datetime, latest_holdings.capital));
468            }
469        }
470        
471        // Update metrics incrementally if in RealTime mode
472        {
473            if let farukon_core::settings::MetricsMode::RealTime { .. } = self.strategy_settings.portfolio_settings_for_strategy.metrics_calculation_mode {
474                let start_date = self.get_all_holdings().first().unwrap().datetime;
475                let end_date = data_handler.get_latest_bar_datetime(
476                    self.strategy_settings.symbols.first().unwrap()
477                ).unwrap();
478
479                // Update deals counter
480                let mut deals_count = 0 as usize;
481                for symbol in &self.strategy_settings.symbols {
482                    if let Some(position_state) = self.current_positions.get_mut(symbol) {
483                        deals_count += position_state.deal_number;
484                    }
485                }
486                
487                if let Some(latest_holdings) = self.get_latest_holdings() {
488                    self.performance_manager.update_incremental(latest_holdings.capital, start_date, end_date, deals_count);
489                    
490                    if self.mode == "Debug".to_string() {
491                        println!(
492                            "Metrics, {:?}",
493                            self.performance_manager.get_current_performance_metrics()
494                        );
495                    }
496                }
497            }
498        }
499        
500        // Margin call monitoring: if capital falls below min_margin, close all positions
501        {
502            let margin_call_monitoring = risks::margin_call_control_for_market(
503                self.get_latest_holdings().unwrap(),
504                self.get_current_positions(),
505                &self.strategy_settings,
506                &self.strategy_instruments_info
507            ).unwrap();
508            if !margin_call_monitoring {
509                for symbol in &self.strategy_settings.symbols {
510                    let quantity = Some(self.get_current_positions().get(symbol).unwrap().position);
511                    if quantity != Some(0.0) {
512                        let _ = self.event_sender.send(Box::new(farukon_core::event::SignalEvent::new(
513                    current_bar_datetime,
514                            symbol.clone(),
515                            "EXIT".to_string(),
516                            "MKT".to_string(),
517                            quantity,
518                            None,
519                        )));
520                    }
521                }
522            }
523        }
524
525        if self.mode == "Debug".to_string() {
526            println!(
527                "Finish event, Update timeindex, {}, {:?}, {:?}",
528                current_bar_datetime,
529                self.current_positions,
530                self.current_holdings,
531            );
532        }
533    }

584    /// Calculates final performance metrics after the backtest ends.
585    fn calculate_final_performance(&mut self) {
586        // Called after backtest ends to compute offline metrics.
587        // Uses full equity curve for accurate drawdown and return calculations.
588
589        if let Some(holdings) = self.get_all_holdings().first() {
590            if let farukon_core::settings::MetricsMode::Offline = self.strategy_settings.portfolio_settings_for_strategy.metrics_calculation_mode {
591                let equity_series = self.get_equity_capital_values();
592                let start_date = holdings.datetime;
593                let end_date = self.get_latest_holdings().unwrap().datetime;
594                
595                let deals_count: usize = self.get_all_positions()
596                    .last()
597                    .map(|snapshot| {
598                        self.strategy_settings.symbols
599                            .iter()
600                            .filter_map(|symbol| snapshot.positions.get(symbol))
601                            .map(|position_state| position_state.deal_number)
602                            .sum()
603                    })
604                    .unwrap_or(0);
605    
606                self.performance_manager.calculate_final(
607                    &equity_series,
608                    start_date,
609                    end_date,
610                    deals_count,
611                );
612            }
613        }
614
615    }
