3//! Dynamic strategy loader: loads compiled Rust libraries (.dylib/.so/.dll) at runtime.
4//! Enables hot-swapping of trading logic without recompiling the core engine.
5//! Uses `libloading` to load symbols: create_strategy, destroy_strategy, calculate_signals.
6
7pub struct DynamicStratagy {
8    _lib: std::sync::Arc<libloading::Library>,  // Holds reference to loaded library
9    strategy_ptr: *mut std::ffi::c_void,    // Pointer to strategy instance
10    destroy_fn: libloading::Symbol<'static, extern "C" fn(*mut std::ffi::c_void)>,  // Destructor
11}
12
13impl DynamicStratagy {
14    pub fn new_from_library(
15        mode: &str,
16        strategy_settings: &farukon_core::settings::StrategySettings,
17        strategy_instruments_info: &std::collections::HashMap<String, farukon_core::instruments_info::InstrumentInfo>,
18        event_sender: &std::sync::mpsc::Sender<Box<dyn farukon_core::event::Event>>,
19        library: std::sync::Arc<libloading::Library>,
20    ) -> anyhow::Result<Self> {
21        let create_strategy: libloading::Symbol<extern "C" fn(
22            *const std::os::raw::c_char,
23            *const farukon_core::settings::StrategySettings,
24            *const std::collections::HashMap<String, farukon_core::instruments_info::InstrumentInfo>,
25            *const std::sync::mpsc::Sender<Box<dyn farukon_core::event::Event>>,
26        ) -> *mut std::ffi::c_void> = unsafe { library.get(b"create_strategy")? };
27
28        let destroy_strategy: libloading::Symbol<extern "C" fn(*mut std::ffi::c_void)> =
29            unsafe { library.get(b"destroy_strategy")? };
30        let mode_c = std::ffi::CString::new(mode)?;
31
32        let strategy_ptr = create_strategy(
33            mode_c.as_ptr(),
34            strategy_settings as *const _,
35            strategy_instruments_info as *const _,
36            event_sender as *const _,
37        );
38
39        if strategy_ptr.is_null() {
40            return Err(anyhow::anyhow!("Failed to create strategy"));
41        }
42
43        let destroy_fn: libloading::Symbol<'static, extern "C" fn(*mut std::ffi::c_void)> = 
44            unsafe { std::mem::transmute(destroy_strategy) };
45
46            anyhow::Ok(DynamicStratagy {
47                _lib: std::sync::Arc::clone(&library),
48                strategy_ptr,
49                destroy_fn,
50            })
51
52    }
53
54    pub fn _load_from_path(
55        mode: &str,
56        strategy_settings: &farukon_core::settings::StrategySettings,
57        strategy_instruments_info: &std::collections::HashMap<String, farukon_core::instruments_info::InstrumentInfo>,
58        event_sender: &std::sync::mpsc::Sender<Box<dyn farukon_core::event::Event>>,
59    ) -> anyhow::Result<Self> {
60        // Loads dynamic strategy library and creates strategy instance.
61        // Expects 3 exported C functions: create_strategy, destroy_strategy, calculate_signals.
62
63        let lib_path = &strategy_settings.strategy_path;
64        let lib = unsafe { libloading::Library::new(lib_path)? };
65
66        let create_strategy: libloading::Symbol<extern "C" fn(
67            *const std::os::raw::c_char,
68            *const farukon_core::settings::StrategySettings,
69            *const std::collections::HashMap<String, farukon_core::instruments_info::InstrumentInfo>,
70            *const std::sync::mpsc::Sender<Box<dyn farukon_core::event::Event>>,
71        ) -> *mut std::ffi::c_void> = unsafe { lib.get(b"create_strategy")? };
72 
73        let destroy_strategy: libloading::Symbol<extern "C" fn(*mut std::ffi::c_void)> =
74            unsafe { lib.get(b"destroy_strategy")? };
75        let mode_c = std::ffi::CString::new(mode)?;
76            
77        let strategy_ptr = create_strategy(
78            mode_c.as_ptr(),
79            strategy_settings as *const _,
80            strategy_instruments_info as *const _,
81            event_sender as *const _,
82        );
83
84        if strategy_ptr.is_null() {
85            return Err(anyhow::anyhow!("Failed to create strategy"));
86        }
87
88        let destroy_fn: libloading::Symbol<'static, extern "C" fn(*mut std::ffi::c_void)> = 
89            unsafe { std::mem::transmute(destroy_strategy) };
90
91        anyhow::Ok(DynamicStratagy {
92            _lib: std::sync::Arc::new(lib),
93            strategy_ptr,
94            destroy_fn,
95        })
96    }
97
98    pub fn calculate_signals(
99        &self,
100        data_handler: &dyn farukon_core::data_handler::DataHandler,
101        current_positions: &std::collections::HashMap<String, farukon_core::portfolio::PositionState>,
102        latest_holdings: &farukon_core::portfolio::HoldingSnapshot,
103        symbol_list: &[String],
104    ) -> anyhow::Result<()> {
105        // Calls calculate_signals() from the loaded library.
106        // Transforms Rust types into C-compatible pointers.
107        // Returns 0 on success, -1 on error.
108
109        let calculate_signals_fn: libloading::Symbol<extern "C" fn (
110            *mut std::ffi::c_void,
111            *const farukon_core::DataHandlerVTable,
112            *const (),
113            *const std::collections::HashMap<String, farukon_core::portfolio::PositionState>,
114            *const farukon_core::portfolio::HoldingSnapshot,
115            *const *const std::os::raw::c_char,
116            usize,
117        ) -> i32> = unsafe {
118            self._lib.get(b"calculate_signals")?
119        };
120
121        let (data_handler_ptr, data_handler_vtable) = unsafe {
122            std::mem::transmute::<_, (*const (), *const farukon_core::DataHandlerVTable)>(data_handler)
123        };
124
125        let c_strings: Vec<std::ffi::CString> = symbol_list
126            .iter()
127            .map(|s| std::ffi::CString::new(s.as_str()))
128            .collect::<anyhow::Result<Vec<_>, _>>()?;
129
130        let c_str_ptrs: Vec<*const std::os::raw::c_char> = c_strings
131            .iter()
132            .map(|s| s.as_ptr())
133            .collect();
134
135        let result = calculate_signals_fn(
136            self.strategy_ptr,
137            data_handler_vtable,
138            data_handler_ptr,
139            current_positions as *const _,
140            latest_holdings as *const _,
141            c_str_ptrs.as_ptr(),
142            symbol_list.len(),
143        );
144
145        if result == 0{
146            anyhow::Ok(())
147        } else {
148            Err(anyhow::anyhow!("Strategy calculate_signals failed with code: {}", result))
149        }
150    }
151
152}
153
154impl Drop for DynamicStratagy {
155    fn drop(&mut self) {
156        // Ensures strategy is destroyed when this object goes out of scope.
157        // Prevents memory leaks.
158
159        if !self.strategy_ptr.is_null() {
160                (self.destroy_fn)(self.strategy_ptr);
161        }
162    }
163}

167unsafe impl Sync for DynamicStratagy {}
