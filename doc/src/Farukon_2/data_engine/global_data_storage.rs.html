5/// Centralized storage for pre-processed market data used by all backtest runs.
6/// This structure is designed to:
7/// - Load raw market data (from FlatBuffers SOA files) once.
8/// - Perform resampling to the target timeframe.
9/// - Align data from all symbols to a single, combined timeline using padding.
10/// - Store the final, ready-to-use Structure-of-Arrays (SOA) data for each symbol.
11/// - Provide a thread-safe, shared reference to this data for multiple backtest instances.
12///
13/// This approach ensures ultra-fast, low-latency access to aligned, zero-copy data,
14/// suitable for SIMD-optimized calculations and event-driven processing.
15#[derive(Debug)]
16pub struct GlobalDataStore {
17    /// Shared reference (Arc) to a HashMap mapping symbol names (e.g., "Si-12.23") to their
18    /// pre-resampled and aligned SOA data.
19    /// Each `SOAData` vector's length corresponds to the length of `combined_timeline`.
20    /// This allows multiple threads (e.g., different optimization runs) to access the same data
21    /// without copying the underlying arrays.
22    soa_data: std::sync::Arc<std::collections::HashMap<String, std::sync::Arc<farukon_core::data_handler::SOAData>>>,
23
24    /// Shared reference (Arc) to the unified, sorted, and deduplicated timeline
25    /// encompassing all timestamps across all symbols after resampling and alignment.
26    /// This serves as the master clock for the backtest, ensuring all symbols are synchronized.
27    combined_timeline: std::sync::Arc<Vec<chrono::DateTime<chrono::Utc>>>,
28
29    /// Flag indicating whether the initial data loading and processing phase has been completed successfully.
30    /// Useful for debugging or checking the readiness of the data store.
31    is_loaded: bool,
32}
33
34impl GlobalDataStore {
35    /// Constructs a new `GlobalDataStore` instance by loading, resampling, and aligning data for all symbols
36    /// specified in the strategy settings.
37    ///
38    /// This method orchestrates the entire data preparation pipeline:
39    /// 1. Loads raw SOA data from FlatBuffer files in parallel.
40    /// 2. Resamples the data to the target timeframe.
41    /// 3. Creates a combined timeline from all resampled data points.
42    /// 4. Aligns and pads each symbol's resampled data to the combined timeline.
43    ///
44    /// # Arguments
45    ///
46    /// * `strategy_settings` - A reference to the `StrategySettings` containing:
47    ///   - `data.data_path`: Directory where `.soa.bin` and `.idx` files are located.
48    ///   - `symbols`: List of symbol names to load data for.
49    ///   - `data.timeframe`: Target timeframe for resampling (e.g., "1min", "5min", "1d").
50    ///   - `threads`: Number of threads to use for parallel processing (optional).
51    ///
52    /// # Returns
53    ///
54    /// * `anyhow::Result<Self>` - On success, returns the initialized `GlobalDataStore`.
55    ///   On failure (e.g., file not found, parsing error), returns an `anyhow::Error`.
56    pub fn load(strategy_settings: &farukon_core::settings::StrategySettings) -> anyhow::Result<Self> {
57        let start = std::time::Instant::now();
58        println!("Starting to load and process FlatBuffer SOA files from {}...", &strategy_settings.data.data_path);
59
60        let resample_timeframe_str = &strategy_settings.data.timeframe;
61        let resample_timeframe_sec = data_engine::utils::resample_timeframe_sec(resample_timeframe_str);
62        let fbs_dir = &strategy_settings.data.data_path;
63        let symbol_list = &strategy_settings.symbols;
64
65        // --- PHASE 1: PARALLEL LOADING AND RESAMPLING ---
66        // Creates a Rayon thread pool to handle loading and resampling of data for multiple symbols concurrently.
67        // This significantly speeds up the initial data preparation phase.
68        let threads = strategy_settings.threads.unwrap_or(num_cpus::get());
69        let pool = rayon::ThreadPoolBuilder::new().num_threads(threads).build()?;
70
71        // Delegates the heavy lifting of loading and resampling to a utility function.
72        // This function uses the created thread pool.
73        // It returns the resampled data per symbol and the partial timelines for each symbol before alignment.
74        let (resampled_data, partial_timelines) = data_engine::utils::load_and_resample(
75            &pool,
76            symbol_list,
77            fbs_dir,
78            resample_timeframe_sec
79        )?;
80        
81        // --- PHASE 2: CREATE COMBINED TIMELINE ---
82        // Delegates the creation of the unified timeline to a utility function.
83        // This function merges and sorts all partial timelines from Phase 1.
84        let combined_timeline_arc = data_engine::utils::create_combined_timeline(
85            &pool,
86            &partial_timelines
87        );
88        let combined_timeline = combined_timeline_arc.as_ref().clone();
89
90        // --- PHASE 3: FINAL PADDING AND ALIGNMENT ---
91        // Delegates the alignment and padding logic to a utility function.
92        // This function takes the resampled data and the unified timeline,
93        // and creates the final `HashMap<String, Arc<SOAData>>` where each `SOAData` is aligned
94        // to the `combined_timeline`.
95        let soa_data = data_engine::utils::align_and_pad(
96            &pool,
97            resampled_data,
98            &combined_timeline
99        )?;
100
101        println!("GlobalDataStore loaded and processed in {:.3} seconds", start.elapsed().as_secs_f64());
102        println!("Symbols loaded: {}", soa_data.len());
103
104        anyhow::Ok(
105            GlobalDataStore {
106                soa_data,
107                combined_timeline: combined_timeline_arc,
108                is_loaded: true,
109            }
110        )
111    }
112
113    pub fn deep_clone(&self) -> Self {
114        let mut new_soa_data = std::collections::HashMap::new();
115        for (symbol, soa_data_arc) in self.soa_data.iter() {
116            let cloned_soa_data = soa_data_arc.deep_clone();
117            new_soa_data.insert(symbol.clone(), std::sync::Arc::new(cloned_soa_data));
118        }
119
120        let new_combined_timeline = self.combined_timeline.as_ref().clone();
121
122        GlobalDataStore {
123            soa_data: std::sync::Arc::new(new_soa_data),
124            combined_timeline: std::sync::Arc::new(new_combined_timeline),
125            is_loaded: self.is_loaded,
126        }
127    }
128
129    /// Provides access to the combined timeline.
130    /// This timeline represents all unique timestamps across all symbols for the backtest period,
131    /// sorted chronologically.
132    ///
133    /// # Returns
134    ///
135    /// * A reference to the internal `Vec<chrono::DateTime<chrono::Utc>>` representing the timeline.
136    pub fn get_combined_timeline(&self) -> &Vec<chrono::DateTime<chrono::Utc>> {
137        &self.combined_timeline
138    }
139
140    /// Provides access to the `is_loaded` flag.
141    ///
142    /// # Returns
143    ///
144    /// * A reference to the boolean flag indicating if the data store has finished loading.
145    pub fn is_loaded(&self) -> &bool {
146        &self.is_loaded
147    }
148
149    /// Provides access to the SOA data for a specific symbol.
150    ///
151    /// # Arguments
152    ///
153    /// * `symbol` - The name of the symbol (e.g., "Si-12.23") to retrieve data for.
154    ///
155    /// # Returns
156    ///
157    /// * `Some(&Arc<SOAData>)` if data for the symbol exists, `None` otherwise.
158    pub fn get_soa_data_for_symbol(&self, symbol: &str) -> Option<&std::sync::Arc<farukon_core::data_handler::SOAData>> {
159        self.soa_data.get(symbol)
160    } 
161
162}
