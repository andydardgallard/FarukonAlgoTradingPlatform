5/// Data handler that provides access to pre-resampled and aligned Structure-of-Arrays (SOA) data.
6/// This handler acts as a bridge between the event-driven backtesting engine and the central `GlobalDataStore`.
7/// It holds a shared reference (`Arc`) to the global store and maintains its own `current_index` to track
8/// the progression of the backtest in the combined timeline. This allows multiple backtest runs
9/// (from parallel optimization) to share the same underlying data source while maintaining
10/// their individual state and preventing look-ahead bias.
11///
12/// The handler implements the `farukon_core::data_handler::DataHandler` trait, providing methods
13/// to retrieve the latest bar's data (OHLCV) for specific symbols based on the current index.
14/// This ensures that strategies only see data up to the current simulated time.
15pub struct SOADataHandlerArc {
16    /// Shared reference to the global data store containing all pre-resampled SOA data for the strategy's symbols.
17    /// This allows multiple `SOADataHandler` instances to access the same zero-copy data.
18    global_data_store: std::sync::Arc<data_engine::global_data_storage::GlobalDataStore>,
19
20    /// The current index in the combined timeline, representing the "current" bar for this specific handler instance.
21    /// Each backtest thread managing its own `SOADataHandler` will increment this index independently,
22    /// ensuring that it only accesses data up to its current time step.
23    current_index: usize,
24    
25    /// Flag indicating whether the backtest should continue.
26    /// Set to `false` when `current_index` reaches the end of the combined timeline.
27    continue_backtest: bool,
28}
29
30impl SOADataHandlerArc {
31    /// Creates a new `SOADataHandler` instance.
32    /// Initializes the handler with the shared `GlobalDataStore` and sets the initial state (index = 0, continue = true).
33    ///
34    /// # Arguments
35    /// * `global_data_store` - An `Arc` wrapped `GlobalDataStore` containing the SOA-formatted, resampled data.
36    ///
37    /// # Returns
38    /// * `SOADataHandler` - The newly created handler instance.
39    pub fn new(global_data_store: std::sync::Arc<data_engine::global_data_storage::GlobalDataStore>) -> Self {
40        
41        // Check if the combined timeline has any data points to determine the initial continue_backtest flag.
42        let continue_backtest = global_data_store.get_combined_timeline().len() > 0;
43        Self {
44            global_data_store,
45            current_index: 0,
46            continue_backtest,
47        }
48    }
49
50    /// Provides access to the current index of this handler.
51    /// This index represents the current position in the `GlobalDataStore`'s combined timeline.
52    ///
53    /// # Returns
54    /// * `usize` - The current index value.
55    pub fn get_current_index(&self) -> usize {
56        self.current_index
57    }
58
59}

61impl farukon_core::data_handler::DataHandler for SOADataHandlerArc {
62    /// Retrieves the latest bar for the specified symbol based on the current index.
63    /// This method constructs a `MarketBar` on-the-fly from the SOA data arrays at the `current_index - 1`.
64    /// It is deprecated in favor of direct value access methods (`get_latest_bar_value`, `get_latest_bar_datetime`)
65    /// for zero-copy performance, but kept for compatibility.
66    ///
67    /// # Arguments
68    /// * `symbol` - The trading symbol (e.g., "Si-12.23").
69    ///
70    /// # Returns
71    /// * `Option<farukon_core::data_handler::MarketBar>` - The latest `MarketBar` or `None` if no data exists or index is 0.
72    fn get_latest_bar(&self, symbol: &str) -> Option<farukon_core::data_handler::SOAData> {
73        // Check if we have processed at least one bar (current_index > 0).
74        if self.current_index > 0 {
75            // Calculate the index of the *previous* bar (the one we just processed).
76            let idx = self.current_index - 1;
77            // Get the SOA data for the specified symbol from the global store.
78            if let Some(soa_data)  = self. global_data_store.get_soa_data_for_symbol(symbol) {
79                // Check if the calculated index is within the bounds of the symbol's data.
80                if idx < soa_data.len() {
81                    let soa_bar = soa_data.get_latest_bar(self.current_index);
82                    Some(soa_bar)
83                } else {
84                    // Index is out of bounds for the symbol's data.
85                    None
86                }
87            } else {
88                // Symbol not found in the global data store.
89                None
90            }
91        } else {
92            // No data has been processed yet (current_index is 0).
93            None
94        }
95    }
96
97    /// Retrieves the last `n` bars for the specified symbol based on the current index.
98    /// This method constructs multiple `MarketBar` structs on-the-fly from the SOA data arrays.
99    /// It is deprecated in favor of `get_latest_bars_values_slice` for zero-copy performance.
100    ///
101    /// # Arguments
102    /// * `symbol` - The trading symbol (e.g., "Si-12.23").
103    /// * `n` - The number of latest bars to retrieve.
104    ///
105    /// # Returns
106    /// * `Option<farukon_core::data_handler::SOAData>` - A `SOAData` struct containing the last `n` bars or `None` if symbol not found.
107    fn get_latest_bars(&self, symbol: &str, n: usize) -> Option<farukon_core::data_handler::SOAData> {
108        let soa_data = if let Some(data) = self.global_data_store.get_soa_data_for_symbol(symbol) {
109            data
110        } else {
111            return None;
112        };
113
114        let start_index = if self.current_index as i64 - n as i64 <= 0 { 0 } else { self.current_index - n };
115        
116        Some(soa_data.set_latest_bars(start_index, self.get_current_index()))
117    }
118
119    /// Retrieves the timestamp of the latest bar for the specified symbol based on the current index.
120    /// This method accesses the combined timeline directly.
121    ///
122    /// # Arguments
123    /// * `symbol` - The trading symbol (e.g., "Si-12.23").
124    ///
125    /// # Returns
126    /// * `Option<chrono::DateTime<chrono::Utc>>` - The timestamp of the latest bar or `None` if no data exists or index is 0.
127    fn get_latest_bar_datetime(&self, symbol: &str) -> Option<chrono::DateTime<chrono::Utc>> {
128        if self.current_index > 0 {
129            let idx = self.current_index - 1;
130            if let Some(soa_data) = self.global_data_store.get_soa_data_for_symbol(symbol) {
131                if idx < soa_data.len() {
132                    let datetime = self.global_data_store.get_combined_timeline()[idx];
133                    Some(datetime) 
134                } else {
135                    // Index is out of bounds for the symbol's data.
136                    None
137                }
138            } else {
139                // Symbol not found in the global data store.
140                None
141            }
142        } else {
143            // No data has been processed yet (current_index is 0).
144            None
145        }
146    }
147
148    /// Retrieves a specific value (open, high, low, close, volume) from the latest bar for the specified symbol based on the current index.
149    /// This is the preferred method for zero-copy access to individual values.
150    ///
151    /// # Arguments
152    /// * `symbol` - The trading symbol (e.g., "Si-12.23").
153    /// * `val_type` - The type of value to retrieve ("open", "high", "low", "close", "volume").
154    ///
155    /// # Returns
156    /// * `Option<f64>` - The requested value or `None` if not available (symbol not found, index out of bounds, or invalid val_type).
157    fn get_latest_bar_value(&self, symbol: &str, val_type: &str) -> Option<f64> {
158        if self.current_index > 0 {
159            let idx = self.current_index - 1;
160            if let Some(soa_data) = self.global_data_store.get_soa_data_for_symbol(symbol) {
161                if idx < soa_data.len() {
162                    match val_type {
163                        "open" => Some(*soa_data.get_open(idx).unwrap()),
164                        "high" => Some(*soa_data.get_high(idx).unwrap()),
165                        "low" => Some(*soa_data.get_low(idx).unwrap()),
166                        "close" => Some(*soa_data.get_close(idx).unwrap()),
167                        "volume" => Some(*soa_data.get_volume(idx).unwrap() as f64),
168                        _ => {
169                            eprintln!("Warning: Unknown value type '{}' for symbol '{}'", val_type, symbol);
170                            None
171                        }
172                    }
173                } else {
174                    None
175                }
176            } else {
177                None
178            }
179        } else {
180            None
181        }
182    }
183
184    /// Retrieves a slice of a specific value (open, high, low, close) from the last `n` bars for the specified symbol based on the current index.
185    /// This is the most efficient way (zero-copy) to access multiple historical values of a single field.
186    ///
187    /// # Arguments
188    /// * `symbol` - The trading symbol (e.g., "Si-12.23").
189    /// * `val_type` - The type of value to retrieve ("open", "high", "low", "close").
190    /// * `n` - The number of latest bars to retrieve values from.
191    ///
192    /// # Returns
193    /// * `Option<&[f64]>` - A slice of the requested values or `None` if symbol not found, index out of bounds, or invalid val_type.
194    fn get_latest_bars_values(&self, symbol: &str, val_type: &str, n: usize) -> Option<&[f64]> {
195        if self.current_index > 0 {
196            let idx = self.current_index - 1;
197            if let Some(soa_data) = self.global_data_store.get_soa_data_for_symbol(symbol) {
198                if idx < soa_data.len() {
199                    let start_index = if self.current_index as i64 - n as i64 <= 0 { 0 } else { self.current_index - n };
200
201                    match val_type {
202                        "open" => Some(&soa_data.get_opens().unwrap()[start_index..self.current_index]),
203                        "high" => Some(&soa_data.get_highs().unwrap()[start_index..self.current_index]),
204                        "low" => Some(&soa_data.get_lows().unwrap()[start_index..self.current_index]),
205                        "close" => Some(&soa_data.get_closes().unwrap()[start_index..self.current_index]),
206                        // Note: This method returns &[f64], so it cannot directly return &[u64] for volume.
207                        // A separate method like get_latest_bars_volume_values would be needed for volume.
208                        _ => {
209                            eprintln!("Warning: Unknown value type '{}' for symbol '{}'", val_type, symbol);
210                            None
211                        }
212                    }
213                } else {
214                    None
215                }
216            } else {
217                None
218            }
219        } else {
220            None
221        }
222    }
223
224    /// Retrieves a slice of volume values (u64) from the last `n` bars for the specified symbol based on the current index.
225    /// This is the most efficient way (zero-copy) to access multiple historical volume values.
226    ///
227    /// # Arguments
228    /// * `symbol` - The trading symbol (e.g., "Si-12.23").
229    /// * `n` - The number of latest bars to retrieve volume values from.
230    ///
231    /// # Returns
232    /// * `Option<&[u64]>` - A slice of the volume values or `None` if symbol not found, index out of bounds.
233    fn get_latest_bars_volume_values(&self, symbol: &str, n: usize) -> Option<&[u64]> {
234        if self.current_index > 0 {
235            let idx = self.current_index - 1;
236            if let Some(soa_data) = self.global_data_store.get_soa_data_for_symbol(symbol) {
237                if idx < soa_data.len() {
238                    let start_index = if self.current_index as i64 - n as i64 <= 0 { 0 } else { self.current_index - n };
239                    Some(&soa_data.get_volumes().unwrap()[start_index..self.current_index])
240                } else {
241                    None
242                }
243            } else {
244                None
245            }
246        } else {
247            None
248        }
249    }

251    /// Advances the internal index to the next bar in the combined timeline.
252    /// This method is called by the `Backtest` loop when a `MARKET` event is received.
253    /// It effectively moves the `SOADataHandler`'s view of "current" data forward in time.
254    fn update_bars(&mut self) {
255        // Check if the current index is within the bounds of the combined timeline.
256        if self.current_index < self.global_data_store.get_combined_timeline().len() {
257            // Increment the current index.
258            self.current_index += 1;
259            // If the index has reached or exceeded the timeline length, set the continue flag to false.
260            if self.current_index >= self.global_data_store.get_combined_timeline().len() {
261                self.continue_backtest = false;
262            }
263        } else {
264            // If the index is already beyond the timeline, set the continue flag to false.
265            self.continue_backtest = false;
266        }
267    }
268
269    /// Checks if the backtest should continue based on the internal state.
270    /// This flag is set to `false` when the `current_index` reaches the end of the combined timeline.
271    ///
272    /// # Returns
273    /// * `bool` - `true` if the backtest should continue, `false` otherwise.
274    fn get_continue_backtest(&self) -> bool {
275        self.continue_backtest
276    }
277
278    /// Sets the flag to stop the backtest.
279    /// This can be used by other components (e.g., `Portfolio` on negative capital) to signal termination.
280    ///
281    /// # Arguments
282    /// * `value` - The new value for the continue flag.
283    fn set_continue_backtest(&mut self, value: bool) {
284        self.continue_backtest = value;
285    }
