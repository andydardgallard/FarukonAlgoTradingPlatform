3/// Structure responsible for simulating order execution.
4/// It mimics broker behavior by applying slippage and commission,
5/// and generates `FILL` events which are sent to the event channel.
6pub struct SimulatedExecutionHandler {
7    /// Channel for sending events (in this case, `FillEvent`).
8    pub event_sender: std::sync::mpsc::Sender<Box<dyn farukon_core::event::Event>>,
9}
10
11impl SimulatedExecutionHandler {
12    /// Creates a new instance of `SimulatedExecutionHandler`.
13    /// # Arguments
14    /// * `event_sender` - The sender for the event channel, which will be used to send `FILL` events.
15    /// # Returns
16    /// * `anyhow::Result<Self>` - A new instance of `SimulatedExecutionHandler` or an error.
17    pub fn new(
18        event_sender: std::sync::mpsc::Sender<Box<dyn farukon_core::event::Event>>,
19    ) -> anyhow::Result<Self> {
20        // Always returns Ok with a new instance, as creating the struct itself cannot fail.
21        anyhow::Ok(
22            Self {
23                event_sender,
24            }
25        )
26    }
27    
28}

30/// Implementation of the `ExecutionHandler` trait from `farukon_core` for `SimulatedExecutionHandler`.
31/// This allows the execution simulator to be used within the main backtesting engine.
32impl farukon_core::execution::ExecutionHandler for SimulatedExecutionHandler {
33    /// The main method that simulates order execution.
34    /// It takes an `ORDER` event, determines the execution price based on the order type (market/limit),
35    /// applies slippage and commission, and sends the result as a `FILL` event.
36    ///
37    /// # Arguments
38    /// * `event` - The `OrderEvent` containing order details (symbol, type, direction, quantity, etc.).
39    /// * `strategy_instruments_info` - A hash-map containing information about instruments traded by the strategy.
40    /// * `strategy_settings` - Strategy settings, including slippage and commission plans.
41    /// * `data_handler` - The data handler used to get the latest bar's price and time.
42    ///
43    /// # Returns
44    /// * `anyhow::Result<()>` - `Ok(())` on success, or an `Err` in case of an error (e.g., missing data).
45    fn execute_order(
46            &self, // Immutable reference to `self` (the struct).
47            event: &farukon_core::event::OrderEvent, // Reference to the order event.
48            strategy_instruments_info: &std::collections::HashMap<String, farukon_core::instruments_info::InstrumentInfo>, // Reference to instrument info map.
49            strategy_settings: &farukon_core::settings::StrategySettings, // Reference to strategy settings.
50            data_handler: &dyn farukon_core::data_handler::DataHandler, // Reference to the data handler (dyn trait object).
51        ) -> anyhow::Result<()> {
52            // Simulates market or limit order execution with slippage and commission.
53            // Uses current bar's high/low for market orders.
54            // For limit orders: checks if the price was hit during the bar.
55            
56            // Get the symbol (e.g., "Si-12.23") from the order event.
57            let symbol = &event.symbol;
58            // Get the instrument information (e.g., exchange, margin, step_price) from the map.
59            let instruments_info = strategy_instruments_info.get(symbol)
60                .ok_or_else(|| anyhow::anyhow!("No instrument info for {}", symbol))?; // Return an error if the information is missing.
61            
62            // Get the date/time of the latest bar for the symbol.
63            let timeindex = data_handler.get_latest_bar_datetime(&event.symbol)
64                .ok_or_else(|| anyhow::anyhow!("Failed to get latest bar datetime for symbol '{}' during order execution", event.symbol))?; // Return an error if date/time is unavailable.
65
66            // Get the data of the latest bar (open, high, low, close, volume).
67            let current_bar = data_handler.get_latest_bar(symbol)
68                .ok_or_else(|| anyhow::anyhow!("No bar for {}", symbol))?; // Return an error if the bar is unavailable.
69
70            // Determine the execution price based on the order type.
71            let execution_price = match event.order_type.as_str() {
72                "MKT" => {
73                    // Market order: slippage is applied.
74                    // Buy at High + slippage, Sell at Low - slippage.
75                    let slippage_val = match &strategy_settings.slippage {
76                        farukon_core::settings::ParamSpec::Discrete(values) => {
77                            if values.len() == 1 {
78                                if let Some(v) = values.first() {
79                                    v.as_f64().ok_or_else(|| anyhow::anyhow!("Slippage value is not a number"))?
80                                } else {
81                                    anyhow::bail!("Slippage vector is empty!");
82                                }
83                            } else {
84                                // If the length of the slippage vector in the strategy settings is not 1, this is an error.
85                                // The current implementation expects only a single slippage value for a market order.
86                                anyhow::bail!("Wrong len of slippage vector!");
87                            }
88                        }
89                        farukon_core::settings::ParamSpec::Range { .. } => {
90                            anyhow::bail!("Slippage cannot be a range in execution!");
91                        }
92                    };
93
94                    match event.direction.as_deref() {
95                        // For a buy order, use the bar's High and add slippage.
96                        Some("BUY") => {
97                            let market_price = (1.0 + slippage_val) * current_bar.get_high(0).unwrap();
98                            if *current_bar.get_low(0).unwrap() <= market_price {
99                                if *current_bar.get_high(0).unwrap() <= market_price {
100                                    *current_bar.get_high(0).unwrap()
101                                } else {
102                                    // If yes, the order is executed at the limit price.
103                                    market_price
104                                }
105                            } else {
106                                // If no, the order is not executed. Return Ok(()) without sending a FillEvent.
107                                return anyhow::Ok(());
108                            }
109                        }
110                        // For a sell order, use the bar's Low and subtract slippage.
111                        Some("SELL") => {
112                            let market_price = (1.0 - slippage_val) * current_bar.get_low(0).unwrap();
113                            if *current_bar.get_high(0).unwrap() >= market_price {
114                                if *current_bar.get_low(0).unwrap() >= market_price {
115                                    *current_bar.get_low(0).unwrap()
116                                } else {
117                                    // If yes, the order is executed at the limit price.
118                                    market_price
119                                }
120                            } else {
121                                // If no, the order is not executed. Return Ok(()) without sending a FillEvent.
122                                return anyhow::Ok(());
123                            }
124                        }
125                        // If the direction is not specified or unknown, return a price of 0.0.
126                        _ => 0.0,
127                    }
128                },
129                "LMT" => {
130                    // Limit order: execution occurs only if the price was reached during the bar.
131                    // Use the specified limit price, or the bar's close price if the limit price is not specified.
132                    let limit_price = event.limit_price.unwrap_or(*current_bar.get_close(0).unwrap());
133                    // Check the direction of the limit order.
134                    match event.direction.as_deref() {
135                        // For a buy order, check if the bar's Low was less than or equal to the limit price.
136                        Some("BUY") => {
137                            if *current_bar.get_low(0).unwrap() <= limit_price {
138                                if *current_bar.get_open(0).unwrap() <= limit_price {
139                                    *current_bar.get_open(0).unwrap()
140                                } else {
141                                    // If yes, the order is executed at the limit price.
142                                    limit_price
143                                }
144                            } else {
145                                // If no, the order is not executed. Return Ok(()) without sending a FillEvent.
146                                return anyhow::Ok(());
147                            }
148                        }
149                        // For a sell order, check if the bar's High was greater than or equal to the limit price.
150                        Some("SELL") => {
151                            if *current_bar.get_high(0).unwrap() >= limit_price {
152                                if *current_bar.get_open(0).unwrap() >= limit_price {
153                                    *current_bar.get_open(0).unwrap()
154                                } else {
155                                    // If yes, the order is executed at the limit price.
156                                    limit_price
157                                }
158                            } else {
159                                // If no, the order is not executed. Return Ok(()) without sending a FillEvent.
160                                return anyhow::Ok(());
161                            }
162                        }
163                        // If the direction is not specified or unknown, the order is not executed.
164                        _ => return anyhow::Ok(()),
165                    }
166                }
167                // If the order type is neither "MKT" nor "LMT", the order is not executed.
168                _ => return anyhow::Ok(()),
169            };
170
171            // Get the exchange name from the instrument information.
172            let exchange = &instruments_info.exchange;
173            // Calculate the commission for the trade using the execution price, instrument info, and strategy settings.
174            let commission = farukon_core::commission_plans::calculate_forts_comission(
175                Some(execution_price), // Pass the execution price.
176                instruments_info,       // Pass the instrument information.
177                strategy_settings       // Pass the strategy settings.
178            );
179            // Calculate the total commission by multiplying the commission per unit by the quantity of contracts.
180            let total_commission = Some(commission.unwrap() * event.quantity);
181            
182            // Create a FillEvent with the details of the executed order.
183            let fill_event = farukon_core::event::FillEvent::new(
184                timeindex,              // Execution time (taken from the current bar).
185                symbol.clone(),         // The instrument symbol.
186                exchange.clone(),       // The exchange name.
187                event.quantity,         // The number of contracts.
188                event.direction.clone(), // The direction of the trade (BUY/SELL).
189                Some(execution_price),  // The price at which the order was executed.
190                total_commission,       // The total commission for the trade.
191                event.signal_name.clone(), // The name of the signal that generated the order.
192            );
193
194            // Send the FillEvent to the event channel.
195            self.event_sender.send(Box::new(fill_event))
196                .map_err(|e| anyhow::anyhow!("Failed to send FillEvent: {}", e))?; // Wrap the send error in an anyhow::Error.
197    
198        // Return Ok(()) upon successful order execution and FillEvent sending.
199        anyhow::Ok(())
200    }
201
