16/// Orchestrates the optimization process for a single strategy.
17/// Manages configuration, runs Grid Search or Genetic Algorithm, and evaluates parameter sets.
18pub struct OptimizationRunner {
19    /// Initial capital allocated to this specific strategy for optimization runs.
20    initial_capital_for_strategy: f64,
21
22    /// Operational mode (e.g., "Debug", "Optimize", "Visual").
23    common_settings: farukon_core::settings::CommonSettings,
24
25    /// Settings specific to the strategy being optimized (parameters, paths, etc.).
26    strategy_settings: farukon_core::settings::StrategySettings,
27
28    /// Instrument metadata (e.g., margin, step, exchange) for all symbols traded by the strategy.
29    strategy_instruments_info: std::collections::HashMap<String, farukon_core::instruments_info::InstrumentInfo>,
30
31    /// The Grid Search optimizer instance, configured based on strategy settings.
32    grid_search_optimizer: farukon_core::optimization::GridSearchOptimizer,
33}

35impl OptimizationRunner {
36    /// Creates a new `OptimizationRunner` instance.
37    /// Initializes the Grid Search optimizer based on the provided strategy settings.
38    /// # Arguments
39    /// * `mode` - The operational mode (affects verbosity and behavior).
40    /// * `initial_capital_for_strategy` - The starting capital for backtests within this optimization run.
41    /// * `strategy_settings` - The configuration for the strategy being optimized.
42    /// * `strategy_instruments_info` - Metadata for all instruments traded by the strategy.
43    /// # Returns
44    /// * `OptimizationRunner` - The newly created runner instance.
45    pub fn new(
46        initial_capital_for_strategy: &f64,
47        common_settings: &farukon_core::settings::CommonSettings,
48        strategy_settings: &farukon_core::settings::StrategySettings,
49        strategy_instruments_info: &std::collections::HashMap<String, farukon_core::instruments_info::InstrumentInfo>
50    ) -> Self {
51        // Parse the strategy settings to extract the ranges for parameters to be optimized.
52        let config = farukon_core::utils::parse_optimization_config(strategy_settings);
53
54        // Create the Grid Search optimizer and configure it with the extracted ranges.
55        let grid_search_optimizer = farukon_core::optimization::GridSearchOptimizer::new()
56            .with_optimization_config(config);
57        
58        Self {
59            initial_capital_for_strategy: *initial_capital_for_strategy,
60            common_settings: common_settings.clone(),
61            strategy_settings: strategy_settings.clone(),
62            strategy_instruments_info: strategy_instruments_info.clone(),
63            grid_search_optimizer,
64        }
65    }

74    /// Executes a Grid Search optimization.
75    /// Evaluates all parameter combinations in parallel using Rayon.
76    /// Each combination triggers a full backtest run.
77    /// # Arguments
78    /// * `total_combinations` - The total number of parameter sets to evaluate.
79    /// * `combinations_to_grid_search` - A vector of `ParameterSet` objects to test.
80    /// # Returns
81    /// * A vector of `OptimizationResult` objects, one for each evaluated parameter set.
82    pub fn run_grid_search(
83        &self,
84        total_combinations: u128,
85        global_data_store: std::sync::Arc<data_engine::global_data_storage::GlobalDataStore>
86    ) -> Vec<farukon_core::optimization::OptimizationResult> {
87        // Runs Grid Search in parallel across all CPU cores.
88        // Each parameter set is evaluated by running a full backtest.
89        // Uses Atomic counter to track progress.
90
91        self.grid_search_optimizer.check_memory_limit(&self.grid_search_optimizer.get_config())
92            .expect("Grid search memory limit exceeded");
93
94        // Determine the number of threads to use for parallel execution.
95        // Defaults to the number of logical CPU cores if not specified in settings.
96        let threads = self.strategy_settings.threads.unwrap_or(num_cpus::get());
97        let global_data_mode = self.common_settings.global_data_storage_mode.clone();
98        let mode = self.common_settings.mode.clone();
99
100        if mode == "Debug" {
101            println!("Starting grid search optimization:");
102            println!("Strategy: {}", self.strategy_settings.strategy_name);
103            println!("Configured threads: {:?}", threads);
104            println!("Total combinations: {}", total_combinations);
105        }
106
107        // Capture necessary data for the parallel execution closure.
108        let initial_capital = self.initial_capital_for_strategy;
109        let strategy_settings = self.strategy_settings.clone();
110        let strategy_instruments_info = self.strategy_instruments_info.clone();
111
112        // Shared atomic counter to track the number of completed evaluations.
113        let counter = std::sync::Arc::new(std::sync::atomic::AtomicUsize::new(0));
114
115        let lib_path = &strategy_settings.strategy_path;
116        let lib = std::sync::Arc::new(
117            unsafe { libloading::Library::new(lib_path)
118                .expect("Can't load library")}
119        );
120
121        // Create a Rayon thread pool with the specified number of threads.
122        let pool = rayon::ThreadPoolBuilder::new()
123            .num_threads(threads)
124            .build()
125            .expect("Failed to create thread pool");
126
127        // Execute the optimization within the thread pool.
128        let results = pool.install(|| {
129            self.grid_search_optimizer.run_optimization(
130                // The fitness function executed for each parameter set.
131                // It runs a backtest and returns an OptimizationResult.
132                move |params| {
133                    let start_time = std::time::Instant::now();
134
135                    // Construct the full parameter set object, including non-strategy parameters like pos_sizer and slippage.
136                    let full_parameter_set = farukon_core::optimization::ParameterSet::new()
137                        .with_strategy_params(params.get_strategy_params().clone())
138                        .with_pos_sizer_name(params.get_pos_sizer_name().clone())
139                        .with_pos_sizer_value(params.get_pos_sizer_value().clone())
140                        .with_pos_sizer_additional_params(params.get_pos_sizer_additional_params().clone())
141                        .with_slippage(params.get_slippage().clone());
142
143                    // Increment the counter and get the current count for logging.
144                    let current_count = counter.fetch_add(1, std::sync::atomic::Ordering::Relaxed) + 1;
145                    println!(
146                        "# {} from {} {}",
147                        current_count,
148                        total_combinations,
149                        full_parameter_set.format_for_display() // Human-readable representation of the parameters.
150                    );
151
152                    // Create temporary strategy settings based on the current parameter set.
153                    let test_settings = farukon_core::utils::create_stratagy_settings_from_params(
154                        &strategy_settings,
155                        &full_parameter_set,
156                    );
157
158                    // Run a full backtest using the current parameter set.
159                    let results = match global_data_mode.as_str() {
160                        "arc" => {
161                            Self::run_backtest_with_settings_arc(
162                                &mode,
163                                &initial_capital,
164                                &test_settings,
165                                &strategy_instruments_info,
166                                std::sync::Arc::clone(&global_data_store),
167                                lib.clone(),
168                            )
169                        }
170                        _ => {
171                            Self::run_backtest_with_settings_deep(
172                                &mode,
173                                &initial_capital,
174                                &test_settings,
175                                &strategy_instruments_info,
176                                std::sync::Arc::clone(&global_data_store),
177                                lib.clone(),
178                            )
179                        }
180                    };
181                    
182                    println!("# {} from {} is done in {:.3} seconds ", current_count, total_combinations, start_time.elapsed().as_secs_f64());
183
184                    // Create an OptimizationResult object containing the parameters and the resulting performance metrics.
185                    farukon_core::optimization::OptimizationResult::new()
186                        .with_parameters(params.clone())
187                        .with_results(results)
188                },
189                threads, // Number of threads to use for the optimization.
190                &self.grid_search_optimizer.get_config(),
191            )
192        });
193
194        results
195    }

197    /// Executes a single backtest run with a given set of strategy parameters.
198    /// This is a helper function used by both Grid Search and Genetic Algorithm.
199    /// # Arguments
200    /// * `mode` - The operational mode.
201    /// * `initial_capital_for_strategy` - The starting capital for this backtest run.
202    /// * `strategy_settings` - The strategy settings, potentially modified with new parameters.
203    /// * `strategy_instruments_info` - Metadata for the instruments traded.
204    /// # Returns
205    /// * `farukon_core::performance::PerformanceMetrics` - The performance metrics from the completed backtest.
206    fn run_backtest_with_settings_arc(
207        mode: &String, 
208        initial_capital_for_strategy: &f64,
209        strategy_settings: &farukon_core::settings::StrategySettings,
210        strategy_instruments_info:  &std::collections::HashMap<String, farukon_core::instruments_info::InstrumentInfo>,
211        global_data_store: std::sync::Arc<data_engine::global_data_storage::GlobalDataStore>,
212        strategy_library: std::sync::Arc<libloading::Library>,
213    ) -> farukon_core::performance::PerformanceMetrics {
214        // Creates a full backtest environment for a single parameter set.
215        // Used by Grid Search and Genetic Algorithm.
216
217        // Create the event channel used for communication between components (DataHandler, Strategy, Portfolio, Execution).
218        let (event_sender, event_receiver) = std::sync::mpsc::channel::<Box<dyn farukon_core::event::Event>>();
219
220        // Initialize the data handler (uses zero-copy FlatBuffers).
221        let data_handler: Box<dyn farukon_core::data_handler::DataHandler> = Box::new(
222            data_engine::data_handler_arc::SOADataHandlerArc::new(global_data_store.clone())
223        );
224
225        // Load the dynamic strategy library (.so/.dylib) specified in settings.
226        let dynamic_strategy: Box<strategy_loader::DynamicStratagy> = Box::new(
227            strategy_loader::DynamicStratagy::new_from_library(
228                mode,
229                strategy_settings,
230                strategy_instruments_info,
231                &event_sender,
232                strategy_library.clone(),
233            ).expect("Failed to load dynamic strategy")
234        );
235
236        // Initialize the portfolio manager.
237        let portfolio: Box<dyn farukon_core::portfolio::PortfolioHandler> = Box::new(
238            portfolio::Portfolio::new(
239                mode,
240                initial_capital_for_strategy,
241                event_sender.clone(),
242                strategy_settings,
243                strategy_instruments_info,
244            ).expect("Failed to create portfolio")
245        );
246
247        // Initialize the simulated execution handler.
248        let execution_handler: Box<dyn farukon_core::execution::ExecutionHandler> = Box::new(
249            execution::SimulatedExecutionHandler::new(
250                event_sender.clone(),
251            ).expect("Failed to create execution handler")
252        );
253
254        // Create the main backtest controller.
255        let mut backtest = backtest::Backtest::new(
256            mode,
257            strategy_settings,
258            strategy_instruments_info,
259            data_handler,
260            event_receiver,
261            dynamic_strategy,
262            portfolio,
263            execution_handler
264        );
265
266        // Run the backtest simulation and return the final performance metrics.
267        backtest.simulate_trading().expect("Backtest failed").clone()
268
269    }

336    /// Executes a Genetic Algorithm optimization.
337    /// Iteratively evolves a population of parameter sets based on their fitness.
338    /// # Arguments
339    /// * `ga_params` - Configuration parameters for the Genetic Algorithm (population size, mutation rate, etc.).
340    /// # Returns
341    /// * `anyhow::Result<Vec<farukon_core::optimization::GAStatsPerGeneration>>` - Statistics for each generation or an error.
342    pub fn run_genetic_search(
343        self,
344        ga_params: &farukon_core::settings::GAParams,
345        global_data_store: std::sync::Arc<data_engine::global_data_storage::GlobalDataStore>,
346    ) -> anyhow::Result<Vec<farukon_core::optimization::GAStatsPerGeneration>> {
347        // Runs Genetic Algorithm optimization.
348        // Uses fitness function to evaluate chromosomes.
349        
350        // Configure the Genetic Algorithm based on the provided parameters.
351        let ga_config = farukon_core::optimization::GAConfig::from_settings(ga_params);
352        // Get the optimization configuration (parameter ranges) for the strategy.
353        let opt_config = self.get_grid_search_optimizer().get_config();
354        // Create the Genetic Algorithm instance.
355        let mut ga = farukon_core::optimization::GeneticAlgorythm::new()
356            .with_ga_config(ga_config.clone())
357            .with_optimization_config(opt_config.clone());
358        
359        let lib_path = &self.strategy_settings.strategy_path;
360        let lib = std::sync::Arc::new(
361            unsafe { libloading::Library::new(lib_path)
362                .expect("Can't load library")
363            }
364        );
365
366        let initial_capital_for_strategy = self.initial_capital_for_strategy;
367        let common_settings = self.common_settings.clone();
368        let strategy_settings = self.strategy_settings.clone();
369        let strategy_instruments_info = self.strategy_instruments_info.clone();
370        let ga_config_closure = ga_config.clone();
371
372        let all_ga_results_shared = std::sync::Arc::new(std::sync::Mutex::new(Vec::new()));
373        let all_ga_results_clone = std::sync::Arc::clone(&all_ga_results_shared);
374
375        // Run the Genetic Algorithm, providing a fitness function that evaluates parameter sets.
376        let stats = ga.run(&strategy_settings.clone(), move |params| {
377            let mode = &common_settings.mode;
378            let global_data_storage_mode = &common_settings.global_data_storage_mode;
379            // Create temporary strategy settings based on the current parameter set for this generation.
380            let test_settings = &farukon_core::utils::create_stratagy_settings_from_params(&strategy_settings, params);
381            // Run a backtest with these parameters.
382            let results = match global_data_storage_mode.as_str() {
383                "arc" => {
384                    Self::run_backtest_with_settings_arc(
385                        mode, 
386                        &initial_capital_for_strategy,
387                        test_settings,
388                        &strategy_instruments_info,
389                        std::sync::Arc::clone(&global_data_store),
390                        lib.clone(),
391                    )
392                }
393                _ => {
394                    Self::run_backtest_with_settings_deep(
395                        mode,
396                        &initial_capital_for_strategy,
397                        test_settings,
398                        &strategy_instruments_info,
399                        std::sync::Arc::clone(&global_data_store),
400                        lib.clone(),
401                    )
402                }
403            };
404            
405            // Create an OptimizationResult object containing the parameters and the resulting performance metrics.
406            let results_obj = farukon_core::optimization::OptimizationResult::new()
407                .with_parameters(params.clone())
408                .with_results(results.clone());
409
410            {
411                let mut results_guard = all_ga_results_clone.lock().unwrap();
412                results_guard.push(results_obj);
413            }
414            Self::calculate_fitness_score(&results, &ga_config_closure)  
415        })?;
416
417        let final_results = {
418            let results_guard = all_ga_results_shared.lock().unwrap();
419            results_guard.clone()
420        };
421
422        // save results
423        self.save_grid_search_optimization_results(&final_results)?;
424        ga.save_generation_stats_to_csv(&stats, &self.strategy_settings)?;
425
426        anyhow::Ok(stats)
427    }

429    /// Calculates a scalar fitness score from performance metrics.
430    /// The score is used by the Genetic Algorithm to rank parameter sets.
431    /// # Arguments
432    /// * `metrics` - The performance metrics from a backtest run.
433    /// * `ga_config` - Configuration specifying the fitness metric and direction (maximize/minimize).
434    /// # Returns
435    /// * `f64` - The calculated fitness score.
436    fn calculate_fitness_score(
437        metrics: &farukon_core::performance::PerformanceMetrics,
438        ga_config: &farukon_core::optimization::GAConfig,
439    ) -> f64 {
440        // Converts performance metrics into a scalar fitness score.
441        // Supports max/min direction and composite metrics.
442
443        // Determine which metric to use for the raw fitness score.
444        let raw_fitness = match ga_config.get_fitness_metric() {
445            farukon_core::settings::FitnessValue::TotalReturn => metrics.get_total_return(),
446            farukon_core::settings::FitnessValue::TotalReturnPercent => metrics.get_total_return_percent(),
447            farukon_core::settings::FitnessValue::APR => metrics.get_apr(),
448            farukon_core::settings::FitnessValue::MaxDD => metrics.get_max_drawdown(),
449            farukon_core::settings::FitnessValue::AprDDFactor => metrics.get_apr_to_drawdown_ratio(),
450            farukon_core::settings::FitnessValue::RecoveryFactor => metrics.get_recovery_factor(),
451            farukon_core::settings::FitnessValue::DealsCount => &(metrics.get_deals_count().clone() as f64),
452            farukon_core::settings::FitnessValue::Composite { metrics: composite_metrics } => {
453                // For composite metrics, calculate a combined score.
454                &Self::calculate_composite_score(metrics, composite_metrics)
455            }
456        };
457
458        // Apply the fitness direction (maximize or minimize).
459        // If direction is "min", the score is negated.
460        let fitness = match ga_config.get_fitness_direction().as_str() {
461            "max" => *raw_fitness,
462            "min" => -raw_fitness,
463            _ => *raw_fitness, // Default to "max" if direction is unknown.
464        };
465
466        fitness
467    }

469    /// Combines multiple performance metrics into a single composite fitness score.
470    /// Each metric contributes equally (simple average).
471    /// # Arguments
472    /// * `metrics` - The performance metrics from a backtest run.
473    /// * `composite_metrics` - A list of metric names to include in the composite score.
474    /// # Returns
475    /// * `f64` - The calculated composite score.
476    fn calculate_composite_score(
477        metrics: &farukon_core::performance::PerformanceMetrics,
478        composite_metrics: &[String],
479    ) -> f64 {
480        // Combines multiple metrics into a single score with equal weights.
481        // Example: APR/DD_factor + Recovery_Factor + Deals_Count
482        
483        let mut total_score = 0.0;
484        // Calculate the weight for each metric (1 / number of metrics).
485        let weight = 1.0 / composite_metrics.len() as f64;
486
487        for metric_name in composite_metrics {
488            // Get the value for the current metric name.
489            let metric_value = match metric_name.as_str() {
490                "Total_Return" => metrics.get_total_return(),
491                "Total_Return_%" => metrics.get_total_return_percent(), 
492                "APR" => metrics.get_apr(),
493                "max_DD" => &(-metrics.get_max_drawdown()),
494                "APR/DD_factor" => metrics.get_apr_to_drawdown_ratio(),
495                "Recovery_Factor" => metrics.get_recovery_factor(),
496                "Deals_Count" => &(-(*metrics.get_deals_count() as f64)), // Negative count for maximization (fewer trades might be better depending on context, but often more is desired, this might need review)
497                _ => &0.0, // Default to 0 if the metric name is unknown.
498            };
499            // Add the weighted value of this metric to the total score.
500            total_score += metric_value * weight;
501        }
502
503        total_score
504    }

506    /// Saves the results of a Grid Search optimization to a CSV file.
507    /// The CSV file contains the tested parameter sets and their corresponding performance metrics.
508    /// This allows for easy analysis and comparison of different hyperparameter combinations.
509    ///
510    /// # Arguments
511    /// * `results` - A slice of `OptimizationResult` objects, each containing a parameter set and its performance metrics.
512    ///
513    /// # Returns
514    /// * `anyhow::Result<()>` - `Ok(())` on success, or an `Err` if file creation or writing fails.
515    pub fn save_grid_search_optimization_results(&self, results: &[farukon_core::optimization::OptimizationResult]) -> anyhow::Result<()> {
516        // --- 1. Determine Output Filename ---
517        // Constructs the path for the output CSV file based on the strategy's settings.
518        // The filename includes a fixed name "optimization_results.csv" appended to the 'exit_results_path'.
519        let filename = format!(
520            "{}/optimization_results.csv",
521            self.strategy_settings.exit_results_path,
522        );
523        let path = std::path::Path::new(&filename);
524
525        // --- 2. Create/Open Output File ---
526        // Attempts to create the file. If it exists, it will be truncated.
527        // Returns an error if the file cannot be created (e.g., directory doesn't exist, permission denied).
528        std::fs::create_dir_all(&self.strategy_settings.exit_results_path)?;
529        let file_exist = path.exists();
530
531        // --- 3. Prepare Header Row (Column Names) ---
532        // --- 3.1: Extract Strategy Parameter Names ---
533        // Gets the names of the strategy-specific parameters that were optimized.
534        // These become the first few column names in the CSV.
535        let mut strategy_params: Vec<String> = self.strategy_settings.strategy_params
536            .iter()
537            .map(|(key, _value)| key.clone())
538            .collect();
539        strategy_params.sort();
540
541        // --- 3.2: Extract Position Sizer Additional Parameter Names ---
542        // Gets the names of any additional parameters for the position sizer (e.g., MPR multiplier).
543        let mut possizers_additional_params: Vec<String> = self.strategy_settings.pos_sizer_params.pos_sizer_params
544            .iter()
545            .map(|(key, _value)| key.clone())
546            .collect();
547        possizers_additional_params.sort();
548
549        // --- 3.3: Extract Performance Metric Names ---
550        // Gets the names of the performance metrics from the *first* result.
551        // Assumes all results have the same set of metrics.
552        // This could be fragile if results have different metrics, but is common for grid search.
553        let result_names = if !file_exist && !results.is_empty() {
554            let mut names: Vec<String> = results[0]
555                .get_results()
556                .to_stats_list()
557                .iter()
558                .map(|(key, _value)| key.clone())
559                .collect();
560            names.sort();
561            Some(names)
562        } else {
563            if !results.is_empty() {
564                let mut names: Vec<String> = results[0]
565                    .get_results()
566                    .to_stats_list()
567                    .iter()
568                    .map(|(key, _value)| key.clone())
569                    .collect();
570                names.sort();
571                Some(names)
572            } else {
573                return anyhow::Ok(());
574            }
575        };
576
577        let mut file = std::fs::OpenOptions::new()
578            .write(true)
579            .append(true)
580            .create(true)
581            .open(&filename)?;
582
583        if !file_exist{
584            if let Some(ref names) = result_names {
585                // --- 3.4: Write Strategy Parameter Column Names ---
586                // Writes the names of the strategy parameters to the file header, separated by semicolons.
587                for name in &strategy_params {
588                        write!(file, "{};", name)?;
589                    }
590
591                // --- 3.5: Write Position Sizer Column Names ---
592                // Adds column names for the position sizing method name and its value.
593                write!(file, "pos_sizer_name;")?;
594                write!(file, "pos_sizer_value;")?;
595
596                // --- 3.6: Write Position Sizer Additional Parameter Column Names ---
597                // Writes the names of the position sizer's additional parameters to the header.
598                for name in &possizers_additional_params {
599                    write!(file, "{};", name)?;
600                }
601        
602                // --- 3.7: Write Slippage Column Name ---
603                // Adds a column name for the slippage parameter used in the test.
604                write!(file, "slippage;")?;
605
606                // --- 3.8: Write Performance Metric Column Names ---
607                // Writes all performance metric names except the last one, followed by a semicolon.
608                // The last metric name is written with a newline character using `writeln!`.element
609                for name in &names[0..names.len() - 1] {
610                    write!(file, "{};", name)?;
611                }
612                // Write the last metric name and add a newline to complete the header row.
613                writeln!(file, "{:?}", names.last().unwrap())?;
614            }
615        }
616
617        // --- 4. Write Data Rows ---
618        // Iterates through each OptimizationResult and writes its parameters and metrics as a row in the CSV.
619        if let Some(ref names) = result_names{
620            for result in results {
621                // --- 4.1: Extract Strategy Parameter Values for Current Result ---
622                // Gets the map of (name, value) for the current result's strategy parameters.
623                let params_map: std::collections::HashMap<_, _> = result
624                    .get_parameters()
625                    .get_strategy_params()
626                    .clone()
627                    .into_iter()
628                    .collect();
629                
630                // Gets the values corresponding to the sorted parameter names, preserving order.
631                let strategy_params_values: Vec<serde_json::Value> = strategy_params
632                    .iter()
633                    .filter_map(|key| params_map.get(key))
634                    .map(|value| value.clone())
635                    .collect();
636    
637                // --- 4.2: Write Strategy Parameter Values ---
638                // Writes each strategy parameter value, followed by a semicolon.
639                for value in &strategy_params_values{
640                    write!(file, "{};", value)?;
641                }
642    
643                // --- 4.3: Write Position Sizer Name and Value ---
644                // Writes the name and value of the position sizing method used for this result.
645                write!(file, "{};", result.get_parameters().get_pos_sizer_name())?;
646                write!(file, "{};", result.get_parameters().get_pos_sizer_value())?;
647                
648                // --- 4.4: Extract Position Sizer Additional Parameter Values ---
649                // Gets the map of (name, value) for the current result's position sizer additional parameters.
650                let pos_sizer_additional_params_map: std::collections::HashMap<_, _> = result
651                    .get_parameters()
652                    .get_pos_sizer_additional_params()
653                    .clone()
654                    .into_iter()
655                    .collect();
656                
657                // Gets the values corresponding to the sorted additional parameter names, preserving order.
658                let pos_sizer_additional_params_values: Vec<serde_json::Value> = possizers_additional_params
659                    .iter()
660                    .filter_map(|key| pos_sizer_additional_params_map.get(key))
661                    .map(|value| value.clone())
662                    .collect();
663                
664                // --- 4.5: Write Position Sizer Additional Parameter Values ---
665                // Writes each position sizer additional parameter value, followed by a semicolon.
666                for value in pos_sizer_additional_params_values {
667                    write!(file, "{};", value)?;
668                }
669    
670                // --- 4.6: Write Slippage Value ---
671                // Writes the slippage value used for this result.
672                write!(file, "{};", result.get_parameters().get_slippage())?;
673    
674                // --- 4.7: Extract Performance Metric Values ---
675                // Gets the map of (metric_name, metric_value_string) for the current result's performance metrics.
676                let performance_metrics_map: std::collections::HashMap<_, _> = result
677                    .get_results()
678                    .to_stats_list()
679                    .into_iter()
680                    .collect();
681    
682
683                // Gets the values corresponding to the sorted metric names, preserving order.
684                let performance_metrics_values: Vec<String> = names
685                    .iter()
686                    .filter_map(|key| performance_metrics_map.get(key))
687                    .map(|value| value.clone())
688                    .collect();
689
690                // --- 4.8: Write Performance Metric Values ---
691                // Writes each performance metric value, followed by a semicolon, except the last one.
692                for value in &performance_metrics_values[0..performance_metrics_values.len() - 1] {
693                    write!(file, "{};", value)?;
694                }
695                writeln!(file, "{}", performance_metrics_values.last().unwrap())?;
696            }
697        }
698
699        // --- 6. Return Success ---
700        // Indicates that the file was written successfully.
701        anyhow::Ok(())
702    }
